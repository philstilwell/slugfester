import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {debates} from '../src/data/debates.js';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1];assert(args.includes('--debate'));assert.match(n,/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p)),hash=b=>createHash('sha256').update(b).digest('hex'),rec=p=>{const b=fs.readFileSync(p);return {path:p,sha256:hash(b),bytes:b.length};};
const registryPath='docs/assessment-production/standalone-debates-v1/registry.json',registry=read(registryPath),r=registry.debates.find(x=>x.debateNumber===n);assert(r);
const root=r.root,dir=root+'/validation/baseline-compatibility-repair',auditPath=r.rhetoricalTagReview.auditPath,correctionPath=r.rhetoricalTagReview.correctionPath,ledgerPath=r.productionLedger.path;
const control='scripts/audit-assessment-production-standalone-rhetorical-tags.mjs',controlText=fs.readFileSync(control,'utf8');
const bounds=controlText.match(/Number\(debate\.number\) >= (\d+) && Number\(debate\.number\) <= (\d+)/);assert(bounds,'Explicit authoritative baseline bounds not found');
const cards=d=>d.sections.flatMap(s=>s.exchanges.flatMap(e=>[e.pro,e.con].filter(Boolean)));
const baseline=debates.filter(d=>+d.number>=+bounds[1]&&+d.number<=+bounds[2]);assert.equal(baseline.length,25);assert(baseline.every(d=>+d.number<+n));
const rows=baseline.map(d=>({number:d.number,moves:cards(d).length,tagged:cards(d).filter(x=>x.tags.length).length,tags:cards(d).reduce((a,x)=>a+x.tags.length,0)}));
const total=k=>rows.reduce((a,x)=>a+x[k],0),counts=rows.map(x=>x.tags).sort((a,b)=>a-b);
const oldAudit=read(auditPath),audit=structuredClone(oldAudit);assert(!audit.recentBaselineComparison);
audit.recentBaselineComparison=structuredClone(audit.baselineComparison);
audit.baselineComparison={debateCount:baseline.length,debateNumbers:baseline.map(d=>d.number),moves:total('moves'),taggedArguments:total('tagged'),tags:total('tags'),taggedArgumentRate:Number((total('tagged')/total('moves')).toFixed(3)),perDebateTagMinimum:counts[0],perDebateTagMedian:counts[Math.floor(counts.length/2)],perDebateTagMaximum:counts.at(-1),zeroTagDebates:rows.filter(x=>x.tags===0).map(x=>x.number),interpretation:oldAudit.baselineComparison.interpretation};
const invariantPaths=[root+'/inventory/inventory.json',root+'/final-ledger/final-ledger.json',root+'/score-pass/output.json',root+'/publication/output.json','src/data/debates.js',r.rhetoricalTagReview.executionPath];
const invariantBefore=invariantPaths.map(rec);
fs.mkdirSync(dir+'/before',{recursive:true});
const targets=[auditPath,correctionPath,ledgerPath,registryPath];
const before=targets.map((p,i)=>{const to=dir+'/before/'+String(i+1).padStart(2,'0')+'-'+p.split('/').at(-1);fs.copyFileSync(p,to,fs.constants.COPYFILE_EXCL);return {original:rec(p),preserved:rec(to)};});
const put=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n');
put(auditPath,audit);
const correction=read(correctionPath);correction.evidence.rhetoricalTagAudit=rec(auditPath);put(correctionPath,correction);
const ledger=read(ledgerPath);assert.equal(ledger.debateId,r.debateId);
ledger.evidenceLocks.publicationRhetoricalTagAudit=rec(auditPath);ledger.evidenceLocks.publicationRhetoricalTagCorrection=rec(correctionPath);put(ledgerPath,ledger);
r.productionLedger=rec(ledgerPath);put(registryPath,registry);
assert.deepEqual(invariantPaths.map(rec),invariantBefore);
for(const k of Object.keys(oldAudit).filter(k=>k!=='baselineComparison'))assert.deepEqual(audit[k],oldAudit[k]);
fs.writeFileSync(dir+'/audit.json',JSON.stringify({status:'passed-deterministic-baseline-metadata-repair',debateNumber:n,debateId:r.debateId,createdAt:new Date().toISOString(),controller:rec('tmp/repair-current-baseline-metadata.mjs'),authoritativeControl:rec(control),reason:'The audit requires its independent historical 25-debate window. The initial controller supplied only a recent window. Both are now retained without changing the validator or any rhetorical decision.',before,after:targets.map(rec),invariants:invariantBefore,changedFields:['rhetorical audit baselineComparison','rhetorical audit recentBaselineComparison','dependent audit hash in rhetorical correction','dependent audit and correction hashes in production adapter','production adapter hash in registry'],judgmentChanges:0,scoreChanges:0,publicationChanges:0,tagChanges:0,testLimitChanges:0,directIncrementalCostUsd:0},null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({status:'baseline-metadata-repaired',historical: audit.baselineComparison.debateNumbers,recent:audit.recentBaselineComparison.debateNumbers,invariantsPreserved:invariantBefore.length}));
