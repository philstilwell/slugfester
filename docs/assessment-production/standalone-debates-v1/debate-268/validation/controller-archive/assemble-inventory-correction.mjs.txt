import fs from 'node:fs';
import assert from 'node:assert/strict';
import {fileRecord} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1];assert(args.includes('--debate'));assert.match(n,/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p)),put=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n',{flag:'wx'});
const rec=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(r=>r.debateNumber===n),root=rec.root,dir=root+'/inventory',p=read(dir+'/correction-1-packet.json'),plan=read(dir+'/correction-1-execution-plan.json'),release=read(dir+'/correction-1-reading-release.json'),c=read(plan.outputPath),old=read(p.priorDraft.path),out=structuredClone(old);
assert.equal(release.status,'authenticated-and-released-for-first-submission');assert.equal(release.outputExistedAtRelease,false);assert(release.inputs.every(i=>i.literalCompleteReadAuthenticated));for(const r of plan.allowedInputs)assert.equal(fileRecord(r.path).sha256,r.sha256);
assert.deepEqual(Object.keys(c).sort(),['replacements','addedMoves','coverageUpdates','selectionBalanceRationale'].sort());assert.deepEqual(c.replacements.map(x=>x.originalMoveId).sort(),p.targetMoveIds.toSorted());assert.equal(new Set(c.replacements.map(x=>x.originalMoveId)).size,p.targetMoveIds.length);assert.equal(c.addedMoves.length,1);
for(const r of c.replacements){const i=out.moves.findIndex(m=>m.moveId===r.originalMoveId);assert(i>=0);const before=old.moves[i];assert.equal(r.move.side,before.side);assert.equal(r.move.speaker,before.speaker);assert.equal(r.move.sectionId,before.sectionId);if(r.originalMoveId!==p.targetMoveIds[2])assert.equal(r.move.moveId,r.originalMoveId);out.moves[i]=r.move;}
const added=c.addedMoves[0],target=old.moves.find(m=>m.moveId===p.targetMoveIds[1]);assert(!old.moves.some(m=>m.moveId===added.moveId));assert.equal(added.side,'pro');assert.equal(added.speaker,old.routes.find(r=>r.side==='pro').speaker);assert.equal(added.sectionId,target.sectionId);assert(added.startEvent>=target.startEvent&&added.endEvent<=target.endEvent);
out.moves.push(added);out.moves.sort((a,b)=>a.startEvent-b.startEvent);
const phases=['symbolic-literal test','independently warranted community evidence and challenge'];assert.deepEqual(c.coverageUpdates.map(x=>x.phase).sort(),phases.toSorted());for(const x of c.coverageUpdates){assert.equal(x.phase,x.row.phase);const i=out.coverageAudit.findIndex(r=>r.phase===x.phase);assert(i>=0);out.coverageAudit[i]=x.row;}
const counts=side=>out.moves.filter(m=>m.side===side).length,sections=out.sections.map(s=>{const pro=out.moves.filter(m=>m.sectionId===s.sectionId&&m.side==='pro').length,con=out.moves.filter(m=>m.sectionId===s.sectionId&&m.side==='con').length;return{sectionId:s.sectionId,pro,con,absoluteDifference:Math.abs(pro-con)};});
assert(typeof c.selectionBalanceRationale==='string'&&c.selectionBalanceRationale.length>=200);
out.selectionBalanceAudit={thresholds:old.selectionBalanceAudit.thresholds,totals:{pro:counts('pro'),con:counts('con'),absoluteDifference:Math.abs(counts('pro')-counts('con'))},sections,rationaleRequired:Math.abs(counts('pro')-counts('con'))>=3||sections.some(s=>s.absoluteDifference>=2),rationale:c.selectionBalanceRationale};
out.publicationCapacityAudit.sections=old.publicationCapacityAudit.sections.map(s=>{const count=sections.find(x=>x.sectionId===s.sectionId);assert(count.pro<=4&&count.con<=4);const fourth=count.pro===4||count.con===4;assert.equal(fourth,s.fourthRowRequired,'New fourth-row need was not authorized in correction');return{...s,pro:count.pro,con:count.con,withinCapacity:true};});
out.speakerAttributionReview.highConfidenceMoveIds=out.moves.filter(m=>m.attributionConfidence==='high').map(m=>m.moveId);out.speakerAttributionReview.belowHigh=out.moves.filter(m=>m.attributionConfidence!=='high').map(m=>({moveId:m.moveId,confidence:m.attributionConfidence,reason:m.audioVerificationReason}));out.speakerAttributionReview.wordingVerification=p.wordingVerification;
for(const m of old.moves.filter(m=>!p.targetMoveIds.includes(m.moveId)))assert.deepEqual(out.moves.find(x=>x.moveId===m.moveId),m,'Unaffected move changed');assert.deepEqual(out.routes,old.routes);assert.deepEqual(out.sections,old.sections);
const es=new Map(read('.assessment-cache/captions/'+rec.videoId+'/scoped-events.json').map(e=>[e.eventIndex,e]));
const review=out.moves.map(m=>{const span=Array.from({length:m.endEvent-m.startEvent+1},(_,i)=>es.get(m.startEvent+i));assert(span.every(Boolean),m.moveId+' ineligible event');assert(span.every(e=>e.attributedSpeaker===m.speaker),m.moveId+' mixed speaker span');const excerpt=span.map(e=>e.text.replace(/\s+/g,' ').trim()).join(' ');assert(excerpt.length<=2200||String(m.sourceSpanSelectionRationale||'').length>=40);for(const q of m.quoteEligibleExactSpans){assert(excerpt.includes(q));assert(q.trim().split(/\s+/).length>=3&&q.trim().split(/\s+/).length<=18);}return{moveId:m.moveId,originalEventRange:[m.startEvent,m.endEvent],speaker:m.speaker,sourceExactQuotes:true,characters:excerpt.length,attributionConfidence:m.attributionConfidence};});
const outputPath=dir+'/corrected-draft-1.json';put(outputPath,out);put(dir+'/correction-1-assembly-audit.json',{schemaVersion:'1.0-bounded-inventory-correction-assembly',status:'passed-authenticated-bounded-correction-pending-semantic-review',createdAt:new Date().toISOString(),debateNumber:n,debateId:rec.debateId,agentPath:release.agentPath,originalDraft:p.priorDraft,correction:fileRecord(plan.outputPath),readingRelease:fileRecord(dir+'/correction-1-reading-release.json'),output:fileRecord(outputPath),allowedSemanticScope:p.targetMoveIds,addedMoveId:added.moveId,unaffectedMovesByteEquivalent:true,routesAndSectionsUnchanged:true,derivedCounts:out.selectionBalanceAudit,capacity:out.publicationCapacityAudit,review,directIncrementalCostUsd:0});console.log(JSON.stringify({status:'assembled-bounded-correction',outputPath,counts:out.selectionBalanceAudit,review},null,2));
