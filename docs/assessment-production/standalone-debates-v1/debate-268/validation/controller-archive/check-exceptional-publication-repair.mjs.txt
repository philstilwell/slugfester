import assert from 'node:assert/strict';
import fs from 'node:fs';
const parts=t=>t.split(/(?=Principal limitation:|Live burden:|Locked score:)/).map(s=>s.trim());
export function checkExceptionalRepair(o,p){
 for(const k of ['debateNumber','debateId'])assert.equal(o[k],p[k]);
 assert.equal(o.status,'complete-authorized-two-field-repair');
 assert.deepEqual(o.fields.map(f=>f.path),p.fields.map(f=>f.path));
 assert.equal(o.fields.length,2);
 for(let i=0;i<2;i++){
  const text=o.fields[i].text,original=p.fields[i].originalText;
  assert.equal(typeof text,'string');const words=text.trim().split(/\s+/).length;
  assert(words>=105&&words<=130,'Critique words '+words);assert(text.length>=880,'Critique characters');
  const after=parts(text),before=parts(original),labels=['Strongest feature:','Principal limitation:','Live burden:','Locked score:'];
  assert.equal(after.length,4);assert(after.every((s,j)=>s.startsWith(labels[j])&&/[.!?]$/.test(s)));
  assert.equal((text.match(/[.!?](?=\s|$)/g)||[]).length,4);
  assert(!/[\u3400-\u9fff\uac00-\ud7af\ufffd]/u.test(text));
  if(i===0)assert.equal(text,original.replace('eastward wind','east wind'));
  else{for(const j of [0,1,3])assert.equal(after[j],before[j]);assert.notEqual(after[2],before[2]);assert(!/audio supplement|expressly forbids|controller|transcription instruction/i.test(after[2]));}
 }
 return{status:'mechanically-valid-source-editorial-review-required',fields:2,unaffectedSentencesPreserved:true};
}
if(process.argv[2]==='--check-stdin')console.log(JSON.stringify(checkExceptionalRepair(JSON.parse(fs.readFileSync(0,'utf8')),JSON.parse(fs.readFileSync(process.argv[3],'utf8')))));
