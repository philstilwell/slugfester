import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {execFileSync} from 'node:child_process';
import {fileRecord,serializedJson,validateStandalonePrimaryJudgment} from './lib/assessment-production-standalone-debate-v1.mjs';
const number=process.argv[process.argv.indexOf('--debate')+1];
assert.match(number||'',/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p));
const registry=read('docs/assessment-production/standalone-debates-v1/registry.json');
const entry=registry.debates.find(r=>r.debateNumber===number);assert.ok(entry);
const root=entry.root,record=fileRecord;
const write=(p,v)=>{assert.equal(fs.existsSync(p),false,`Refuse overwrite ${p}`);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,serializedJson(v),{flag:'wx'});};
const authorizationPath=`${root}/recovery/pass-b-replacement-1-authorization.json`,authorization=read(authorizationPath);
const planPath=`${root}/judgments/replacement-pass-b-1/execution-plan.json`,plan=read(planPath);
const reviewPath=`${root}/judgments/replacement-pass-b-1/controller-content-review.json`,review=read(reviewPath);
assert.equal(authorization.debateId,entry.debateId);assert.equal(plan.debateId,entry.debateId);
assert.equal(review.status,'passed-source-specific-content-review');
assert.equal(review.output.sha256,record(plan.outputPath).sha256);
assert.equal(review.allDimensionRationalesRead,true);
assert.equal(review.burdenRationalesRead,true);
assert.equal(authorization.additionalAttemptsAuthorized,1);assert.equal(plan.attemptLimit,1);
for(const r of authorization.preserved)assert.equal(record(r.path).sha256,r.sha256,`Changed original ${r.path}`);
for(const r of plan.allowedInputs)assert.equal(record(r.path).sha256,r.sha256,`Changed allowed input ${r.path}`);
const inventoryPath=`${root}/inventory/inventory.json`,inventory=read(inventoryPath),inventorySha256=record(inventoryPath).sha256;
const replacement=read(plan.outputPath);
validateStandalonePrimaryJudgment(replacement,inventory,{expectedPass:'pass-b',expectedInventorySha256:inventorySha256});
validateStandalonePrimaryJudgment(read(`${root}/judgments/pass-a/output.json`),inventory,{expectedPass:'pass-a',expectedInventorySha256:inventorySha256});
assert.equal(review.dimensionRationalesReviewed,inventory.moves.length*6);
const archiveRoot=`${root}/recovery/initial-failed-cohort`;
const canonicalPaths=[`${root}/judgments/pass-b/output.json`,`${root}/judgments/execution.json`,`${root}/disagreements/disagreements.json`,`${root}/adjudication/packet.json`];
const preserved=canonicalPaths.map(p=>{
 const destination=`${archiveRoot}/${p.slice(root.length+1)}`;
 fs.mkdirSync(path.dirname(destination),{recursive:true});fs.copyFileSync(p,destination,fs.constants.COPYFILE_EXCL);
 const original=record(p),archive=record(destination);assert.equal(original.sha256,archive.sha256);return {original,archive};
});
write(`${archiveRoot}/preservation.json`,{status:'exact-original-bytes-preserved-before-activation',debateNumber:number,debateId:entry.debateId,preservedAt:new Date().toISOString(),records:preserved,notUsableForScoring:true,reason:'Initial second assessment passed structural checks but failed source-specific content review; preliminary disagreements derived from that rejected assessment are quarantined.'});
const originalExecution=read(canonicalPaths[1]);
const newExecution=structuredClone(originalExecution);
newExecution.completedAt=new Date().toISOString();
Object.assign(newExecution.execution,{simultaneousLaunch:false,recoveryMode:'user-authorized-single-fresh-whole-pass-replacement',isolationMethod:'Retained original isolated pass-a plus one independently isolated replacement pass-b using the exact same frozen score-blind evidence packet; initial pass-b and preliminary downstream cohort preserved separately',attemptCountPerPass:{'pass-a':1,'pass-b':2},retryCountPerPass:{'pass-a':0,'pass-b':0},automaticRetries:0,additionalReplacementAttempts:1});
newExecution.passes[1]={pass:'pass-b',outputPath:canonicalPaths[0],outputSha256:record(plan.outputPath).sha256,outputBytes:record(plan.outputPath).bytes,judgmentCount:inventory.moves.length,validationStatus:'passed',validationError:null,attemptCount:2,retryCount:0,replacementAttemptCount:1,writablePaths:[plan.outputPath],rawReplacementOutput:record(plan.outputPath),agentTask:plan.agentTask};
newExecution.audit={...newExecution.audit,primaryContexts:3,acceptedPrimaryContexts:2,schemaValidPasses:2,sourceSpecificContentValidPasses:2,rejectedInitialSecondPassPreserved:true,passAUnchanged:true,sourceUnchanged:true,inventoryUnchanged:true};
newExecution.recovery={authorization:record(authorizationPath),plan:record(planPath),contentReview:record(reviewPath),preservedFailureCohort:record(`${archiveRoot}/preservation.json`)};
const replacementStage=`${canonicalPaths[0]}.replacement-stage`;
fs.copyFileSync(plan.outputPath,replacementStage,fs.constants.COPYFILE_EXCL);fs.renameSync(replacementStage,canonicalPaths[0]);
const executionStage=`${canonicalPaths[1]}.replacement-stage`;write(executionStage,newExecution);fs.renameSync(executionStage,canonicalPaths[1]);
// These invalid preliminary files remain exactly recoverable in the authenticated archive.
for(const p of canonicalPaths.slice(2)){assert.equal(record(p).sha256,preserved.find(r=>r.original.path===p).archive.sha256);fs.unlinkSync(p);}
console.log(execFileSync(process.execPath,['scripts/audit-assessment-production-standalone-v1.mjs','--write-disagreements','--debate',number],{encoding:'utf8',maxBuffer:5*1024*1024}));
write(`${root}/recovery/replacement-activation.json`,{status:'replacement-activated-and-disputes-rebuilt',debateNumber:number,debateId:entry.debateId,completedAt:new Date().toISOString(),authorization:record(authorizationPath),rawReplacement:record(plan.outputPath),canonicalReplacement:record(canonicalPaths[0]),execution:record(canonicalPaths[1]),retainedPassA:record(`${root}/judgments/pass-a/output.json`),unchangedInventory:record(inventoryPath),preservedFailureCohort:record(`${archiveRoot}/preservation.json`),newDisagreements:record(canonicalPaths[2]),newAnonymousPacket:record(canonicalPaths[3]),directIncrementalCostUsd:0});
console.log(JSON.stringify({debateNumber:number,acceptedJudgments:2,retainedFirstPass:true,newDisputes:read(canonicalPaths[2]).disputes.length}));
