import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {fileRecord,serializedJson,canonicalJson,validateStandaloneCandidate} from './lib/assessment-production-standalone-debate-v1.mjs';
const arg=k=>process.argv[process.argv.indexOf(k)+1],number=arg('--debate'),phase=arg('--phase');assert.match(number||'',/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p)),registry=read('docs/assessment-production/standalone-debates-v1/registry.json'),entry=registry.debates.find(x=>x.debateNumber===number);assert.ok(entry);
const root=entry.root,rr=`${root}/publication/length-repair-1`,rawPath=`${root}/publication/raw-prose-output-1.json`,raw=read(rawPath);
assert.equal(raw.debateNumber,number);assert.equal(raw.debateId,entry.debateId);
const write=(p,v)=>{assert(!fs.existsSync(p),`Refuse overwrite ${p}`);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,serializedJson(v),{flag:'wx'});};
const wc=s=>s.trim().split(/\s+/u).filter(Boolean).length;
const fields=raw.candidate.sections.flatMap((s,si)=>s.exchanges.flatMap((e,ei)=>['pro','con'].flatMap(side=>e[side]?[{path:`candidate.sections[${si}].exchanges[${ei}].${side}.critique`,sectionIndex:si,exchangeIndex:ei,side,moveId:e[side].ledgerMoveId,text:e[side].critique,score:e[side].score}]:[])));
const failures=fields.filter(f=>wc(f.text)>130);
const model={slug:'gpt-5.6-sol',label:'5.6 Sol',reasoningEffort:'low',authentication:'ChatGPT subscription'};
if(phase==='freeze'){
 const initial=read(`${root}/publication/initial-validation.json`),ledger=read(`${root}/final-ledger/final-ledger.json`);assert.equal(initial.output.sha256,fileRecord(rawPath).sha256);
 assert.equal(initial.errors.length,failures.length);assert(initial.errors.every(e=>e.includes('critique words')));
 const shards=[];
 for(let i=0;i<failures.length;i+=2){const ordinal=shards.length+1,id=String(ordinal).padStart(2,'0'),selected=failures.slice(i,i+2),inputPath=`${rr}/shard-${id}-packet.json`,outputPath=`${rr}/shard-${id}-output.json`;
  const packet={schemaVersion:'1.0-bounded-publication-length-repair',status:'frozen-before-execution',debateNumber:number,debateId:entry.debateId,ordinal,motion:raw.candidate.motion,model,sourcePublication:fileRecord(rawPath),validation:fileRecord(`${root}/publication/initial-validation.json`),frozenLedger:fileRecord(`${root}/final-ledger/final-ledger.json`),fields:selected.map(f=>({...f,originalWords:wc(f.text),evidence:ledger.moves.find(m=>m.moveId===f.moveId)})),instructions:{maximumWritableLeafFields:2,oneSubmission:true,retries:0,words:[105,130],targetWords:[122,126],minimumCharacters:880,exactlyFourSentences:true,labels:['Strongest feature:','Principal limitation:','Live burden:','Locked score:'],preserveMeaningQualificationsAndScore:true,scope:'Shorten only the listed critique strings; do not add claims, tags, or new reasoning. Preserve the source-specific praise, limitation, live burden, and score interpretation. No other file or field is writable.',validateInMemoryBeforeSubmission:true},outputContract:{schemaVersion:'1.0',status:'complete-and-schema-valid',debateNumber:number,ordinal,repairs:'Exact fields array order [{path,moveId,text}] with only permitted revised critique strings.'},outputPath};
  write(inputPath,packet);shards.push({ordinal,agentTask:`/root/prose_${number}_repair_${id}`,input:fileRecord(inputPath),outputPath,writableFieldCount:selected.length,writablePaths:selected.map(f=>f.path),attempts:1,retries:0});
 }
 write(`${rr}/execution-plan.json`,{schemaVersion:'1.0',status:'frozen-before-isolated-length-repairs',debateNumber:number,debateId:entry.debateId,model,freshContextPerShard:true,forkTurns:'none',sourcePublication:fileRecord(rawPath),failureCount:failures.length,shards,directIncrementalCostUsd:0});console.log(JSON.stringify({fields:failures.length,shards:shards.length}));
}else if(phase==='apply'){
 const plan=read(`${rr}/execution-plan.json`),next=structuredClone(raw);assert.equal(plan.sourcePublication.sha256,fileRecord(rawPath).sha256);const checks=[];
 for(const s of plan.shards){assert.equal(s.input.sha256,fileRecord(s.input.path).sha256);const p=read(s.input.path),o=read(s.outputPath);assert.equal(o.schemaVersion,'1.0');assert.equal(o.status,'complete-and-schema-valid');assert.equal(o.debateNumber,number);assert.equal(o.ordinal,s.ordinal);assert.deepEqual(o.repairs.map(r=>r.path),s.writablePaths);assert.equal(o.repairs.length,s.writableFieldCount);
  o.repairs.forEach((r,i)=>{const f=p.fields[i];assert.equal(r.moveId,f.moveId);assert(wc(r.text)>=105&&wc(r.text)<=130);assert(r.text.length>=880);const labels=p.instructions.labels,parts=r.text.split(/(?=Principal limitation:|Live burden:|Locked score:)/).map(s=>s.trim());assert.equal(parts.length,4);assert(parts.every((s,i)=>s.startsWith(labels[i])&&/[.!?]$/.test(s)));assert.equal((r.text.match(/[.!?](?=\s|$)/g)||[]).length,4);assert(r.text.includes(String(f.score)));assert.equal(next.candidate.sections[f.sectionIndex].exchanges[f.exchangeIndex][f.side].critique,f.text);next.candidate.sections[f.sectionIndex].exchanges[f.exchangeIndex][f.side].critique=r.text;checks.push({path:r.path,moveId:r.moveId,words:wc(r.text),characters:r.text.length});});
 }
 const stripped=structuredClone(next);for(const f of failures)stripped.candidate.sections[f.sectionIndex].exchanges[f.exchangeIndex][f.side].critique=f.text;assert.equal(canonicalJson(stripped),canonicalJson(raw));validateStandaloneCandidate(next.candidate,read(`${root}/score-pass/output.json`));
 write(`${root}/publication/output.json`,next);write(`${rr}/validation.json`,{status:'passed-bounded-length-repairs',source:fileRecord(rawPath),after:fileRecord(`${root}/publication/output.json`),checks,onlyDeclaredFieldsChanged:true});
 write(`${rr}/execution.json`,{schemaVersion:'1.0-bounded-publication-repair-execution',status:'passed-isolated-card-depth-repair',debateNumber:number,debateId:entry.debateId,model,freshContextPerShard:true,forkTurns:'none',shards:plan.shards.map(s=>({...s,output:fileRecord(s.outputPath)})),audit:{judgmentChanges:0,scoreChanges:0,moveChanges:0,tagChanges:0,maximumWritableFieldsPerShard:2},directIncrementalCostUsd:0});console.log(JSON.stringify({status:'passed',repairedFields:checks.length,output:fileRecord(`${root}/publication/output.json`)}));
}else throw new Error('Unknown phase');
