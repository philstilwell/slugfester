import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {execFileSync} from 'node:child_process';
import {fileRecord,serializedJson,validateStandalonePrimaryJudgment} from './lib/assessment-production-standalone-debate-v1.mjs';
const number=process.argv[process.argv.indexOf('--debate')+1];assert.match(number||'',/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p));
const entry=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(d=>d.debateNumber===number);assert.ok(entry);
const root=entry.root,ip=`${root}/inventory/inventory.json`,inventory=read(ip),inventorySha256=fileRecord(ip).sha256;
const replacementSequence=2;
const replacementRoot=`${root}/judgments/pass-a-replacement-${replacementSequence}`;
const approvalPath=`${root}/recovery/approved-recovery-${replacementSequence}.json`;
const planPath=`${replacementRoot}/execution-plan.json`,plan=read(planPath),approval=read(approvalPath);
assert.equal(approval.status,'explicitly-authorized-by-user');assert.equal(approval.maximumReplacementAttempts,1);
assert.equal(approval.replacementSequence,replacementSequence);
assert.equal(plan.replacementSequence,replacementSequence);
assert.equal(fileRecord(approvalPath).sha256,plan.approval.sha256);
for(const failure of approval.priorFailedAttempts)assert.equal(fileRecord(failure.path).sha256,failure.sha256);
for(const r of plan.allowedInputs)assert.equal(fileRecord(r.path).sha256,r.sha256);
assert.equal(fileRecord(plan.retainedPassB.path).sha256,plan.retainedPassB.sha256);
const replacementPath=plan.writablePaths[0],a=read(replacementPath),bPath=`${root}/judgments/pass-b/output.json`,b=read(bPath);
validateStandalonePrimaryJudgment(a,inventory,{expectedPass:'pass-a',expectedInventorySha256:inventorySha256});
validateStandalonePrimaryJudgment(b,inventory,{expectedPass:'pass-b',expectedInventorySha256:inventorySha256});
const rootReview=read(`${replacementRoot}/controller-validation.json`);
assert.equal(rootReview.status,'passed-and-preserved');assert.equal(rootReview.output.sha256,fileRecord(replacementPath).sha256);assert.equal(rootReview.rationaleReview.controllerReadEveryRationale,true);
const write=(p,o)=>{assert.equal(fs.existsSync(p),false);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,serializedJson(o),{flag:'wx'});};
const canonicalPath=`${root}/judgments/pass-a/output.json`;
assert.equal(fs.existsSync(canonicalPath),false);fs.mkdirSync(path.dirname(canonicalPath),{recursive:true});fs.copyFileSync(replacementPath,canonicalPath,fs.constants.COPYFILE_EXCL);
assert.equal(fileRecord(canonicalPath).sha256,fileRecord(replacementPath).sha256);
write(`${replacementRoot}/execution.json`,{status:'complete-and-schema-valid',debateNumber:number,debateId:entry.debateId,agentTask:plan.agentTask,model:plan.model,displayModel:plan.displayModel,reasoningEffort:plan.reasoningEffort,authentication:plan.authentication,freshContext:true,forkTurns:'none',plan:fileRecord(planPath),rawOutput:fileRecord(replacementPath),canonicalOutput:fileRecord(canonicalPath),controllerReview:fileRecord(`${replacementRoot}/controller-validation.json`),attempts:1,retries:0,priorFailedPrimaryAttempts:approval.priorFailedAttempts.length,mechanicalByteCopyOnly:true,directIncrementalCostUsd:0});
const passes=['pass-a','pass-b'].map((pass,i)=>{const outputPath=`${root}/judgments/${pass}/output.json`,r=fileRecord(outputPath);return {pass,outputPath,outputSha256:r.sha256,outputBytes:r.bytes,judgmentCount:inventory.moves.length,validationStatus:'passed',validationError:null,attemptCount:i?1:1+replacementSequence,retryCount:i?0:replacementSequence,automaticRetryCount:0,explicitlyAuthorizedReplacementCount:i?0:replacementSequence,writablePaths:[i?outputPath:replacementPath],agentTask:i?`/root/judgment_${number}_b`:plan.agentTask};});
write(`${root}/judgments/execution.json`,{schemaVersion:'1.1-standalone-judgment-execution',protocolVersion:'standalone-debate-v1',debateNumber:Number(number),debateId:entry.debateId,status:'complete-and-schema-valid',execution:{model:plan.model,displayModel:plan.displayModel,reasoningEffort:plan.reasoningEffort,billingSurface:plan.authentication,freshContextPerPass:true,forkTurns:'none',simultaneousLaunch:false,isolationMethod:'Original pass-b retained; the second expressly authorized fresh built-in isolated replacement supplies pass-a using the identical frozen score-blind packet and a separate direct-JSON raw output path. Both failed pass-a preparations remain preserved.',sharedPacketPath:plan.sharedPacket.path,judgmentPacketSha256:plan.sharedPacket.sha256,inventorySha256,allowedInputsIdentical:true,outputsDisjoint:true,initialPrimaryContexts:2,additionalReplacementContexts:replacementSequence,totalAssessmentContexts:2+replacementSequence,schemaValidPrimaryContexts:2,automaticRetries:0,directIncrementalCostUsd:0},passes,recovery:{approval:fileRecord(approvalPath),priorApproval:approval.priorApproval,failedAttempts:approval.priorFailedAttempts,replacementExecution:fileRecord(`${replacementRoot}/execution.json`),retainedPassB:plan.retainedPassB},completedAt:new Date().toISOString(),audit:{primaryContexts:2+replacementSequence,schemaValidPasses:2,identicalEvidencePackets:true,mutuallyBlind:true,calculatedScoresSeenByModels:false,publicationSeenByModels:false,rerunPermitted:false}});
console.log(execFileSync(process.execPath,['scripts/audit-assessment-production-standalone-v1.mjs','--write-disagreements','--debate',number],{encoding:'utf8',maxBuffer:8*1024*1024}));
