import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {fileRecord,serializedJson} from './lib/assessment-production-standalone-debate-v1.mjs';
const number=process.argv[process.argv.indexOf('--debate')+1];
assert.match(number||'',/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p));
const entry=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(d=>d.debateNumber===number);assert.ok(entry);
const root=entry.root,manifest=read(`${root}/manifest.json`),packetPath=`${root}/judgments/judgment-packet.json`,packet=read(packetPath);
for(const r of [...Object.values(manifest.sourceLocks),...Object.values(manifest.controlLocks)])assert.equal(fileRecord(r.path).sha256,r.sha256);
const prior=read(`${root}/judgments/pass-b/controller-validation.json`);
assert.equal(fileRecord(prior.output.path).sha256,prior.output.sha256);
assert.equal(fs.existsSync(`${root}/judgments/pass-a/output.json`),false);
const write=(p,o)=>{assert.equal(fs.existsSync(p),false);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,serializedJson(o),{flag:'wx'});};
const authorization={status:'explicitly-authorized-by-user',debateNumber:number,debateId:entry.debateId,videoId:entry.videoId,userReply:'Proceed with both.',scope:'One fresh isolated replacement for failed pass-a, retaining pass-b and frozen source evidence; paid audio verification up to $4.50, expected approximately $0.48, one attempt per clip and no paid retries.',authorizedAt:new Date().toISOString()};
write(`${root}/recovery/approved-recovery-1.json`,{...authorization,failedAttempt:fileRecord(`${root}/recovery/pass-a-preparation-failure/diagnostic.json`),retainedPassB:prior.output,judgmentPacket:fileRecord(packetPath),maximumReplacementAttempts:1,maximumAudioCostUsd:4.5,sourceInventoryAndValidJudgmentImmutable:true});
write(`${root}/audio/cost-authorization-exception.json`,{...authorization,plan:fileRecord(`${root}/audio/plan.json`),maximumUsd:4.5,expectedUsd:read(`${root}/audio/plan.json`).cost.estimatedUsd,standingMaximumUsd:1,maximumAttemptsPerClip:1,automaticRetries:0,approval:fileRecord(`${root}/recovery/approved-recovery-1.json`)});
write(`${root}/judgments/pass-a-replacement-1/execution-plan.json`,{status:'frozen-before-execution',debateNumber:number,debateId:entry.debateId,pass:'pass-a',model:packet.assessmentModel==='5.6 Sol'?'gpt-5.6-sol':null,displayModel:packet.assessmentModel,reasoningEffort:packet.reasoningEffort,authentication:packet.authentication,agentTask:`/root/judgment_${number}_a_replacement_1`,freshContext:true,forkTurns:'none',sharedPacket:fileRecord(packetPath),allowedInputs:[fileRecord(packetPath),...Object.values(packet.evidenceLocks)],writablePaths:[`${root}/judgments/pass-a-replacement-1/output.json`],attemptLimit:1,retries:0,approval:fileRecord(`${root}/recovery/approved-recovery-1.json`),retainedPassB:prior.output,failedDraftAndOtherReviewForbidden:true,calculatedScoresForbidden:true,directIncrementalCostUsd:0});
console.log(JSON.stringify({debateNumber:number,validPassBPreserved:true,replacementPlan:`${root}/judgments/pass-a-replacement-1/execution-plan.json`,audioMaximumUsd:4.5}));
