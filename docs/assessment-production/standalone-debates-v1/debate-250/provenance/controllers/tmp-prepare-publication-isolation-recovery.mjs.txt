import fs from 'node:fs';
import assert from 'node:assert/strict';
import path from 'node:path';
import {fileRecord, serializedJson, canonicalJson} from './lib/assessment-production-standalone-debate-v1.mjs';
const number=process.argv[process.argv.indexOf('--debate')+1];assert.match(number||'',/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p)),record=p=>fileRecord(p);
const registry=read('docs/assessment-production/standalone-debates-v1/registry.json');
const entry=registry.debates.find(d=>d.debateNumber===number);assert.ok(entry);
const root=entry.root,rr=`${root}/publication/isolation-recovery-1`,originalPlan=read(`${root}/publication/execution-plan.json`);
assert.equal(originalPlan.debateId,entry.debateId);assert.equal(originalPlan.attemptLimit,1);
for(const p of originalPlan.writablePaths)assert.equal(fs.existsSync(p),false,'A publication output already exists; do not replace it');
for(const p of originalPlan.allowedInputs)assert.equal(record(p.path).sha256,p.sha256);
const write=(p,v)=>{assert.equal(fs.existsSync(p),false);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,serializedJson(v),{flag:'wx'});};
const ledgerPath=`${root}/final-ledger/final-ledger.json`,scorePath=`${root}/score-pass/output.json`;
const ledger=read(ledgerPath),scores=read(scorePath),ledgerBefore=record(ledgerPath),scoresBefore=record(scorePath);
assert.equal(ledger.debateId,entry.debateId);assert.equal(scores.debateId,entry.debateId);
write(`${rr}/failure-record.json`,{status:'publication-context-stopped-before-output',debateNumber:number,debateId:entry.debateId,agentTask:originalPlan.agentTask,executionPlan:record(`${root}/publication/execution-plan.json`),cause:'The allowed resolved final ledger embeds original pass-a and pass-b judgments. The publication writer read that first move before recognizing conflict with its no-raw-primary-judgments instruction.',workerMessage:'No output written. All 13 input hashes verified. Isolation breach: the allowed final ledger’s first move exposed embedded primary judgments during inspection. A fresh publication context should receive a filtered ledger view excluding `judgments`.',outputsWritten:0,sourceChanges:0,acceptedJudgmentChanges:0,scoreChanges:0,directIncrementalCostUsd:0,replacementExecuted:false,requiresExplicitUserAuthorization:true,recordedAt:new Date().toISOString()});
const {evidenceLocks:unusedEvidence,moves,...body}=ledger;
const filtered={...body,schemaVersion:'1.0-publication-resolved-evidence-view',status:'derived-read-only-publication-view',moves:moves.map(({judgments,...move})=>move)};
assert.equal(filtered.moves.length,ledger.moves.length);
for(let i=0;i<moves.length;i++){const {judgments,...expected}=moves[i];assert.equal(canonicalJson(filtered.moves[i]),canonicalJson(expected));}
assert.equal(/"(?:judgments|passA|passB)"/.test(JSON.stringify(filtered)),false);
write(`${rr}/resolved-evidence.json`,filtered);
const {scoreStability,...finalScores}=scores;
write(`${rr}/calculated-scores.json`,finalScores);
const packet=read(`${root}/publication/packet.json`),originalTemplate=canonicalJson(packet.template);
packet.evidenceLocks.finalLedger=record(`${rr}/resolved-evidence.json`);
packet.evidenceLocks.scores=record(`${rr}/calculated-scores.json`);
assert.equal(canonicalJson(packet.template),originalTemplate);
write(`${rr}/packet.json`,packet);
const replace=new Map([[`${root}/publication/packet.json`,`${rr}/packet.json`],[ledgerPath,`${rr}/resolved-evidence.json`],[scorePath,`${rr}/calculated-scores.json`]]);
const inputs=originalPlan.allowedInputs.map(r=>record(replace.get(r.path)??r.path));
assert.ok(inputs.every(r=>r.path!==ledgerPath&&r.path!==scorePath));
write(`${rr}/proposed-execution-plan.json`,{...originalPlan,status:'prepared-awaiting-explicit-user-authorization',agentTask:`/root/publication_${number}_replacement_1`,allowedInputs:inputs,writablePaths:[`${rr}/raw-prose-output.json`],attemptLimit:1,retries:0,executionPermitted:false,restriction:'Read only the listed views, source and controls. Original final ledger, original calculated-score output, original publication packet, all primary judgments and other debates remain unavailable. The original final-ledger path may be cited in sourceQuoteAudit without opening it.',failure:record(`${rr}/failure-record.json`),additionalDirectIncrementalCostEstimateUsd:0});
assert.deepEqual(record(ledgerPath),ledgerBefore);assert.deepEqual(record(scorePath),scoresBefore);
write(`${rr}/filter-validation.json`,{status:'passed-mechanical-view-filter',debateNumber:number,debateId:entry.debateId,originalLedger:ledgerBefore,filteredLedger:record(`${rr}/resolved-evidence.json`),originalScores:scoresBefore,filteredScores:record(`${rr}/calculated-scores.json`),removedFields:['ledger.evidenceLocks','ledger.moves[*].judgments','scores.scoreStability'],preservedMoveCount:moves.length,allOtherMoveFieldsByteEquivalent:true,templateUnchanged:true,originalArtifactsUnchanged:true,modelCalls:0,controller:record('scripts/tmp-prepare-publication-isolation-recovery.mjs')});
console.log(JSON.stringify({status:'prepared-not-authorized-or-executed',moves:moves.length,originalLedgerUnchanged:true,scoreUnchanged:true}));
