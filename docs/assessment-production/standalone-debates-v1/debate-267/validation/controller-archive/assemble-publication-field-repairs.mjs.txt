import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {checkRepair} from './check-publication-field-repair.mjs';import {checkPublication} from './check-publication-draft.mjs';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1];assert(args.includes('--debate'));assert.match(n,/^\d{3,}$/);const read=p=>JSON.parse(fs.readFileSync(p)),rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:createHash('sha256').update(b).digest('hex'),bytes:b.length};},put=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n',{flag:'wx'});

assert.deepEqual(seen,plan.fields);
const quoteAudit=['pro','con'].map(side=>{const text=candidate.quotes[side].text,matches=ledger.moves.filter(m=>m.side===side&&m.quoteEligibleExactSpans.some(s=>s.includes(text)));assert.equal(matches.length,1,'Ambiguous exact quotation requires review');const m=matches[0];return{side,moveId:m.moveId,speaker:candidate.sides[side].speaker,text,quoteEligibleExactSpan:m.quoteEligibleExactSpans.find(s=>s.includes(text)),sourcePath:bundle.ledgerPath,transcriptPath:bundle.transcriptPath,...Object.fromEntries(['startEvent','endEvent','startMs','endMs'].map(k=>[k,m.sourceSpan[k]])),exactMatch:true};});
const novelty=read(base+'/controller-novelty-map.json');assert.equal(novelty.status,'passed-direct-source-specific-novelty-review');assert.equal(novelty.aiRepairOutputSha256,rec(base+'/ai-contribution/output.json').sha256);
const output={...bundle.outputRequirements,candidate,sourceQuoteAudit:quoteAudit,noveltyMap:novelty.entries,audit:bundle.outputRequirements.audit,publicationRecovery:{plan:rec(base+'/plan.json'),initialFailure:rec(dir+'/initial-failure/diagnostic.json'),initialSubmittedOutputs:0,repairGroups:contexts.length,repairShards:plan.shards,maximumWritableLeafFieldsPerShard:2,attemptsPerShard:1,recursiveRetries:0,mechanicalAssemblyOnly:true}};
const validation=checkPublication(output,bundle);put(dir+'/output.json',output);
const execution={schemaVersion:'1.1-isolated-publication-with-unavailable-field-recovery',status:'passed-authenticated-isolated-publication',debateNumber:n,debateId:r.debateId,model:read(dir+'/execution-plan.json').model,initialContext:{executionPlan:rec(dir+'/execution-plan.json'),readingRelease:rec(dir+'/reading-release.json'),failure:rec(dir+'/initial-failure/diagnostic.json'),submittedOutputs:0},repairPlan:rec(base+'/plan.json'),repairContexts:contexts.map(x=>({groupId:x.groupId,agentPath:x.agentPath,model:x.model,actualModelContexts:x.actualModelContexts,execution:rec(base+'/'+x.groupId+'/execution.json'),output:x.output,shards:x.shards})),output:rec(dir+'/output.json'),validation,mechanicalAssembly:true,initialAttempts:1,repairAttemptsPerShard:1,repairShards:plan.shards,retriesOfExhaustedRepairs:0,scoreChanges:0,judgmentChanges:0,directIncrementalCostUsd:0};put(dir+'/execution.json',execution);put(base+'/assembly-audit.json',{status:'passed-bounded-field-repair-mechanical-assembly',debateNumber:n,debateId:r.debateId,plan:rec(base+'/plan.json'),output:rec(dir+'/output.json'),execution:rec(dir+'/execution.json'),fieldsApplied:seen,shards:plan.shards,sourceQuoteAuditDerived:true,noveltyMap:rec(base+'/controller-novelty-map.json'),validation,judgmentChanges:0,scoreChanges:0});console.log(JSON.stringify({status:execution.status,repairGroups:contexts.length,repairShards:plan.shards,validation}));
