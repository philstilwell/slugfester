import fs from 'node:fs';
import assert from 'node:assert/strict';
import {fileRecord,sha256,validateStandaloneInventory} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';

const changes=[];

const caps=sections.map(s=>{const prior=d.publicationCapacityAudit.sections.find(x=>x.sectionId===s.sectionId),fourthRowRequired=s.pro===4||s.con===4;return{sectionId:s.sectionId,pro:s.pro,con:s.con,withinCapacity:s.pro<=4&&s.con<=4,fourthRowRequired,fourthRowAuthorized:!fourthRowRequired||(prior.fourthRowAuthorized===true&&prior.rationale.length>=40),rationale:prior.rationale};});

console.log(JSON.stringify({status:'correction-validated',validation,scopeValidation,counts:d.selectionBalanceAudit,capacity:d.publicationCapacityAudit,preserved:args.includes('--preserve')},null,2));
