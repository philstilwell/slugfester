import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {checkRhetorical} from './check-rhetorical-draft.mjs';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1],phase=args[args.indexOf('--phase')+1];assert(args.includes('--debate')&&args.includes('--phase'));assert.match(n,/^\d{3,}$/);assert(['pass-a','pass-b','adjudication'].includes(phase));
const read=p=>JSON.parse(fs.readFileSync(p)),hash=b=>createHash('sha256').update(b).digest('hex'),rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:hash(b),bytes:b.length};};
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(r=>r.debateNumber===n);assert(r);const dir=r.root+'/publication/rhetorical-tag-review-1/'+phase,plan=read(dir+'/execution-plan.json'),release=read(dir+'/reading-release.json');
assert.equal(release.status,'authenticated-and-released-for-first-submission');assert.equal(release.outputExistedAtRelease,false);assert(release.inputs.every(i=>i.literalCompleteReadAuthenticated));for(const i of plan.allowedInputs)assert.equal(rec(i.path).sha256,i.sha256);
const rows=fs.readFileSync(release.session.path,'utf8').trim().split('\n').map(JSON.parse),contexts=rows.filter(r=>r.type==='turn_context').map(r=>({model:r.payload.model,reasoningEffort:r.payload.effort}));assert(contexts.length&&contexts.every(c=>c.model===plan.model.slug&&c.reasoningEffort===plan.model.reasoningEffort));
const validation=checkRhetorical(read(plan.outputPath),read(plan.packetPath));
const audit={schemaVersion:'1.0-isolated-rhetorical-context-execution',status:'passed-authenticated-rhetorical-'+phase,debateNumber:n,debateId:r.debateId,agentPath:release.agentPath,model:plan.model,actualModelContexts:contexts,session:{...rec(release.session.path),kind:'prefix-snapshot-after-output'},executionPlan:rec(dir+'/execution-plan.json'),readingRelease:rec(dir+'/reading-release.json'),readingAudit:rec(plan.readingAuditPath),input:rec(plan.packetPath),output:rec(plan.outputPath),validation,attempts:1,retries:0,protocolIsolation:true,operatingSystemFilesystemIsolation:false,directIncrementalCostUsd:0};
fs.writeFileSync(dir+'/execution.json',JSON.stringify(audit,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({status:audit.status,validation},null,2));
