import fs from 'node:fs';
import assert from 'node:assert/strict';
import {pipeline} from 'node:stream/promises';
import {createHash} from 'node:crypto';
const planPath=process.argv[2];
assert(planPath, 'Explicit source authorization path required');
const plan=JSON.parse(fs.readFileSync(planPath));
assert.equal(plan.status,'user-approved-limited-scope-awaiting-source-verification');
const id=plan.videoId, cache=`.assessment-cache/captions/${id}`, out=`${cache}/original-audio.m4a`;
const caption=JSON.parse(fs.readFileSync(`${cache}/manifest.json`));
assert.equal(caption.videoId,id); assert.equal(caption.sourceUrl,plan.canonicalUrl); assert(!fs.existsSync(out));
const html=await(await fetch(plan.canonicalUrl,{signal:AbortSignal.timeout(30000)})).text();
const key=html.match(/"INNERTUBE_API_KEY":"([^"]+)"/)?.[1]; assert(key);
const response=await fetch(`https://www.youtube.com/youtubei/v1/player?key=${key}`,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({context:{client:{clientName:'ANDROID',clientVersion:'20.10.38',androidSdkVersion:30,hl:'en',gl:'US'}},videoId:id}),signal:AbortSignal.timeout(30000)});
assert(response.ok,`Player status ${response.status}`);
const player=await response.json();
assert.equal(player.videoDetails?.videoId,id); assert.equal(player.videoDetails?.title,caption.title);
assert.equal(Number(player.videoDetails.lengthSeconds),caption.durationSeconds);
const candidates=(player.streamingData?.adaptiveFormats??[]).filter(f=>f.itag===140&&f.url&&!f.audioTrack?.isAutoDubbed);
assert.equal(candidates.length,1,'Expected one public default original audio stream');
const format=candidates[0]; assert(!format.audioTrack||format.audioTrack.audioIsDefault);
const media=await fetch(format.url,{signal:AbortSignal.timeout(120000)});
assert(media.ok,`Audio fetch HTTP ${media.status}`);
await pipeline(media.body,fs.createWriteStream(out,{flags:'wx'}));
const bytes=fs.readFileSync(out);
const record={videoId:id,canonicalUrl:plan.canonicalUrl,retrievedAt:new Date().toISOString(),acquisition:'Public YouTube Android player; single default original audio stream',format:{itag:format.itag,mimeType:format.mimeType,audioTrack:format.audioTrack??null,approxDurationMs:format.approxDurationMs},file:{path:out,bytes:bytes.length,sha256:createHash('sha256').update(bytes).digest('hex')},directCostUsd:0};
fs.writeFileSync(`${out}.provenance.json`,JSON.stringify(record,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify(record,null,2));
