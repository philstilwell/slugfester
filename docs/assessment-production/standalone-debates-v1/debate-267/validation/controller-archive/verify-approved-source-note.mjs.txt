import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
const args = process.argv.slice(2);
assert.equal(args[0], '--debate');
const n = args[1];
assert.match(n, /^\d{3,}$/);
const read = p => JSON.parse(fs.readFileSync(p));
const registry = read('docs/assessment-production/standalone-debates-v1/registry.json');
const r = registry.debates.find(x => x.debateNumber === n);
assert(r);
const root = r.root + '/publication';
const correction = read(root + '/source-disclosure-review/approved-correction.json');
assert.equal(correction.debateId, r.debateId);
assert.deepEqual(correction.writableFields, ['candidate.sourceNote']);
assert.equal(correction.authorization.additionalAttemptsAuthorized, 1);
assert.equal(correction.authorization.additionalAttemptsUsed, 1);
const priorBytes = fs.readFileSync(correction.preservedPriorOutput.path);
const repairedBytes = fs.readFileSync(root + '/rhetorical-tag-review-1/preserved-prior-output.json');
const hash = b => createHash('sha256').update(b).digest('hex');
assert.equal(hash(priorBytes), correction.preservedPriorOutput.sha256);
assert.equal(hash(repairedBytes), correction.after.sha256);
const before = JSON.parse(priorBytes), after = JSON.parse(repairedBytes);
assert.equal(after.candidate.sourceNote, before.candidate.sourceNote + correction.appendedText);
assert.equal(read(root + '/output.json').candidate.sourceNote, after.candidate.sourceNote);
after.candidate.sourceNote = before.candidate.sourceNote;
assert.deepEqual(after, before);
console.log('Approved source-note repair verified: one exact appended sentence, one additional attempt, all other pre-tag fields unchanged.');
