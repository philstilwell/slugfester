import fs from 'node:fs';
import assert from 'node:assert/strict';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1];assert(args.includes('--debate'));assert.match(n,/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p)),r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
const c=read(r.root+'/publication/output.json').candidate,cards=c.sections.flatMap(s=>s.exchanges.flatMap(e=>[e.pro,e.con].filter(Boolean)));
const expected={number:n,title:c.title,motion:c.motion,cards:cards.length,tags:cards.flatMap(c=>c.tags).length,firstRole:cards[0].role,url:c.youtubeUrl,disclosure:read(r.root+'/source/reader-scope-disclosure.json').requiredReaderDisclosure,speakers:['pro','con'].map(s=>c.sides[s].speaker)};
const body=async(page)=>{
 const width=page.viewportSize().width,label=width===390?'mobile':'desktop',check=(v,m)=>{if(!v)throw Error(m);};
 const cards=page.locator('article.argument');check(await cards.count()===expected.cards,'Locked card count');
 check((await cards.locator(':scope > p').allTextContents()).every(t=>t.trim().length>30),'Empty argument');
 check(await cards.locator('a.tag').count()===expected.tags,'Card tags');
 check((await page.getByRole('heading',{level:1}).innerText()).includes(expected.title),'Title');
 const scope=page.getByRole('region',{name:'Source and assessment scope'});check((await scope.innerText()).includes(expected.disclosure),'Exact scope');
 check(!(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth)),'Horizontal overflow');
 await scope.scrollIntoViewIfNeeded();await page.screenshot({path:`output/playwright/debate-${expected.number}/${label}-scope.png`});
 const summary=page.locator('details.ai-extension-accordion > summary'),details=page.locator('details.ai-extension-accordion');
 check(!await details.evaluate(el=>el.open),'Default collapsed');await summary.scrollIntoViewIfNeeded();await page.screenshot({path:`output/playwright/debate-${expected.number}/${label}-ai-closed.png`});
 await summary.click();check(await details.evaluate(el=>el.open),'Pointer open');await summary.press('Enter');check(!await details.evaluate(el=>el.open),'Enter close');await summary.press('Space');check(await details.evaluate(el=>el.open),'Space open');
 await page.screenshot({path:`output/playwright/debate-${expected.number}/${label}-ai-open.png`});await summary.click();
 const button=page.getByRole('button',{name:'Critique for '+expected.firstRole,exact:true}).first();await button.scrollIntoViewIfNeeded();await button.click();
 const tipId=await button.getAttribute('aria-describedby'),tip=page.locator(`[id="${tipId}"]`);
 await page.waitForFunction(id=>{const s=getComputedStyle(document.getElementById(id));return s.visibility==='visible'&&Number(s.opacity)===1;},tipId);
 const visible=()=>tip.evaluate(el=>{const s=getComputedStyle(el);return s.visibility!=='hidden'&&Number(s.opacity)>0&&s.display!=='none';});
 check(await visible(),'Critique pointer');await page.mouse.move(0,0);await button.focus();await button.press('Enter');check(await visible(),'Critique Enter');await button.press('Space');check(await visible(),'Critique Space');check(await button.evaluate(el=>el===document.activeElement),'Focus retained');
 await page.screenshot({path:`output/playwright/debate-${expected.number}/${label}-critique.png`});await summary.click({timeout:5000});check(await details.evaluate(el=>el.open),'Distant control intercepted');await summary.click();
 const portraits=await page.locator('main img[src*="/assets/caricatures/"]').evaluateAll(es=>es.map(e=>({src:e.getAttribute('src'),loaded:e.complete&&e.naturalWidth>0})));check(portraits.length>=2&&portraits.every(p=>p.loaded),'Portraits loaded');
 check(!(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth)),'Final overflow');
 console.log(JSON.stringify({status:'passed-debate-interactions',viewport:page.viewportSize(),cards:expected.cards,cardTags:expected.tags,scopeVisible:true,aiPointerEnterSpace:true,critiquePointerEnterSpaceFocus:true,distantControlUnintercepted:true,portraits,horizontalOverflow:false}));
};
const out='output/playwright/debate-'+n;fs.mkdirSync(out,{recursive:true});fs.writeFileSync(out+'/check-debate-ui.js','async (page) => { const expected='+JSON.stringify(expected)+'; return ('+body.toString()+')(page); }\n',{flag:'wx'});
console.log(JSON.stringify({status:'browser-check-derived-from-frozen-publication',debateNumber:n,output:out+'/check-debate-ui.js',expected}));
