import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';
import {validateStandalonePrimaryJudgment} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1];assert(args.includes('--debate'));assert.match(n,/^\d{3,}$/);const read=p=>JSON.parse(fs.readFileSync(p)),hash=b=>createHash('sha256').update(b).digest('hex');
const rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:hash(b),bytes:b.length};};
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert.ok(r);const root=r.root;
const inv=read(root+'/inventory/inventory.json'),invHash=rec(root+'/inventory/inventory.json').sha256,plan=read(root+'/judgments/execution-plan.json');assert.equal(plan.execution.inventorySha256,invHash);
const passes=[];
for(const pass of ['pass-a','pass-b']){
 const dir=root+'/judgments/'+pass,release=read(dir+'/reading-release.json'),p=read(dir+'/execution-instructions.json'),output=read(p.outputPath);
 assert.equal(release.status,'authenticated-and-released-for-first-submission');assert.equal(release.outputExistedAtRelease,false);assert.ok(release.inputs.every(x=>x.literalCompleteReadAuthenticated));
 for(const input of p.allowedInputs)assert.equal(rec(input.path).sha256,input.sha256);
 const validation=validateStandalonePrimaryJudgment(output,inv,{expectedPass:pass,expectedInventorySha256:invHash});
 const raw=fs.readFileSync(release.session.path),rows=raw.toString().trim().split('\n').map(JSON.parse),contexts=rows.filter(r=>r.type==='turn_context').map(r=>({model:r.payload.model,reasoningEffort:r.payload.effort}));
 assert.ok(contexts.length&&contexts.every(x=>x.model===plan.execution.model&&x.reasoningEffort===plan.execution.reasoningEffort));
 const rationales=output.judgments.flatMap(j=>Object.values(j.dimensions).map(d=>d.rationale));
 assert.equal(rationales.length,inv.moves.length*6);assert.equal(new Set(rationales).size,rationales.length,'Repeated complete rationale requires manual review, not automatic acceptance.');
 const authentication={schemaVersion:'1.0-primary-execution-authentication',status:'schema-source-and-model-authenticated',debateNumber:n,debateId:r.debateId,pass,modelContexts:contexts,session:{path:release.session.path,sha256:hash(raw),bytes:raw.length,kind:'prefix-snapshot-after-output'},readingRelease:rec(dir+'/reading-release.json'),readingAudit:rec(p.readingAuditPath),instructions:rec(dir+'/execution-instructions.json'),output:rec(p.outputPath),validation,attempts:1,retries:0,dimensionRationales:rationales.length,uniqueCompleteRationales:new Set(rationales).size,minimumRationaleCharacters:Math.min(...rationales.map(s=>s.length)),protocolIsolation:true,operatingSystemFilesystemIsolation:false,directIncrementalCostUsd:0};
 fs.writeFileSync(dir+'/session-authentication.json',JSON.stringify(authentication,null,2)+'\n',{flag:'wx'});
 passes.push({pass,outputPath:p.outputPath,outputSha256:rec(p.outputPath).sha256,judgmentCount:output.judgments.length,validationStatus:'passed',readingRelease:authentication.readingRelease,sessionAuthentication:rec(dir+'/session-authentication.json'),attempts:1,retries:0,directIncrementalCostUsd:0});
}
const execution={schemaVersion:'1.0-standalone-judgment-execution',status:'complete-and-schema-valid',debateNumber:Number(n),debateId:r.debateId,execution:plan.execution,executionPlan:rec(root+'/judgments/execution-plan.json'),passes,directIncrementalCostUsd:0};
fs.writeFileSync(root+'/judgments/execution.json',JSON.stringify(execution,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({status:execution.status,passes:passes.map(p=>({pass:p.pass,judgmentCount:p.judgmentCount,validationStatus:p.validationStatus}))},null,2));
