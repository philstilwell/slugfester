import fs from 'node:fs';
import assert from 'node:assert/strict';
import {fileRecord,validateStandaloneCandidate} from './scripts/lib/assessment-production-standalone-debate-v1.mjs';
const [flag,number,extra,candidateArg]=process.argv.slice(2);assert.equal(flag,'--debate');if(extra)assert.equal(extra,'--candidate');
const read=p=>JSON.parse(fs.readFileSync(p));
const record=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===number);assert(record);
const root=record.root,outputPath=candidateArg??`${root}/publication/output.json`,out=read(outputPath),c=out.candidate,scores=read(`${root}/score-pass/output.json`),ledger=read(`${root}/final-ledger/final-ledger.json`),skeleton=read(`${root}/publication/input-bundle.json`).lockedCandidateSkeleton;
const errors=[],check=(ok,path,message)=>{if(!ok)errors.push({path,message});},wc=s=>String(s??'').trim().split(/\s+/).filter(Boolean).length,bounds=(s,path,min,max=Infinity)=>check(typeof s==='string'&&wc(s)>=min&&wc(s)<=max,path,`Needs ${min}–${max} words; found ${wc(s)}`);
const fixed=(a,b,p='candidate')=>{if(typeof b==='string'&&b==='')return;if(Array.isArray(b)){check(Array.isArray(a)&&a.length===b.length,p,'Frozen array structure changed');b.forEach((v,i)=>fixed(a?.[i],v,`${p}[${i}]`));}else if(b&&typeof b==='object'){check(a&&typeof a==='object',p,'Frozen object missing');Object.entries(b).forEach(([k,v])=>fixed(a?.[k],v,`${p}.${k}`));}else check(a===b,p,'Frozen value changed');};fixed(c,skeleton);
check(out.schemaVersion==='1.0-standalone-publication-output'&&out.protocolId==='assessment-production-standalone-debate-v1'&&out.status==='complete-publication-prose-candidate'&&out.debateNumber===number&&out.debateId===record.debateId,'output','Envelope mismatch');
try{validateStandaloneCandidate(c,scores);}catch(e){errors.push({path:'candidate',message:e.message});}
bounds(c.summary,'candidate.summary',8,35);bounds(c.sourceNote,'candidate.sourceNote',10);bounds(c.scoringNote,'candidate.scoringNote',18);check(/AI-generated/i.test(c.scoringNote),'candidate.scoringNote','Explicit AI-generated disclosure missing');
const cards=c.sections.flatMap((s,si)=>s.exchanges.flatMap((e,ei)=>['pro','con'].flatMap(side=>e[side]?[{...e[side],side,path:`candidate.sections[${si}].exchanges[${ei}].${side}`}]:[]))),labels=['Strongest feature:','Principal limitation:','Live burden:','Locked score:'];
for(const m of cards){bounds(m.words,`${m.path}.words`,8,55);bounds(m.role,`${m.path}.role`,1,5);bounds(m.critique,`${m.path}.critique`,105,130);check(m.critique.length>=880,`${m.path}.critique`,`Needs >=880 characters; found ${m.critique.length}`);let parts=m.critique.split(/(?=Principal limitation:|Live burden:|Locked score:)/).map(s=>s.trim());check(parts.length===4&&parts.every((s,i)=>s.startsWith(labels[i])&&/[.!?]$/.test(s))&&(m.critique.match(/[.!?](?=\s|$)/g)||[]).length===4,`${m.path}.critique`,'Exactly four ordered labeled sentences required');check(!/[\u3400-\u9fff\uac00-\ud7af\ufffd]/u.test(m.critique),`${m.path}.critique`,'Unexpected script artifact');check(m.tags.length===0,`${m.path}.tags`,'Tags must await independent review');}
const descriptionMean=cards.reduce((s,m)=>s+wc(m.words),0)/cards.length;check(descriptionMean>=20.9,'candidate.argumentDescriptions','Mean below independent benchmark20.9');check(cards.filter(m=>wc(m.words)<20).length/cards.length<=.25,'candidate.argumentDescriptions','Too many short cards');
for(const side of ['pro','con']){
 const q=c.quotes[side];bounds(q.text,`candidate.quotes.${side}.text`,3,18);bounds(q.context,`candidate.quotes.${side}.context`,12,55);
 const qa=out.sourceQuoteAudit?.find(x=>x.side===side),m=ledger.moves.find(x=>x.moveId===qa?.moveId);check(Boolean(m)&&m.side===side&&qa.speaker===m.speaker&&qa.text===q.text&&qa.exactMatch===true&&qa.quoteEligibleExactSpan.includes(q.text)&&m.quoteEligibleExactSpans.includes(qa.quoteEligibleExactSpan),`candidate.quotes.${side}.text`,'Must be exact eligible substring attributed to correct side');
 if(m){for(const k of ['startEvent','endEvent','startMs','endMs'])check(qa[k]===m.sourceSpan[k],`sourceQuoteAudit.${side}.${k}`,'Source coordinate differs');}
 c.overall[side].strengths.forEach((s,i)=>bounds(s,`candidate.overall.${side}.strengths[${i}]`,1));c.overall[side].blunders.forEach((b,i)=>{bounds(b.text,`candidate.overall.${side}.blunders[${i}].text`,8);check(b.links.length===0,`candidate.overall.${side}.blunders[${i}].links`,'Links must await independent review');});
 const ex=c.logicalExtension[side],p=`candidate.logicalExtension.${side}`;bounds(ex.finalArgument.thesis,`${p}.finalArgument.thesis`,12);ex.finalArgument.premises.forEach((s,i)=>bounds(s,`${p}.finalArgument.premises[${i}]`,12));bounds(ex.finalArgument.conclusion,`${p}.finalArgument.conclusion`,15);ex.newArguments.forEach((a,i)=>{bounds(a.title,`${p}.newArguments[${i}].title`,2,8);bounds(a.text,`${p}.newArguments[${i}].text`,45,130);});
 const fields=['finalArgument',...ex.newArguments.map((_,i)=>`newArguments[${i}]`)];for(const f of fields){const rows=out.noveltyMap?.filter(x=>x.side===side&&x.field===f)||[];check(rows.length===1&&rows[0].path===`${p}.${f}`&&rows[0].hypothetical===true&&Array.isArray(rows[0].sourceMoveIds)&&rows[0].sourceMoveIds.every(id=>ledger.moves.some(m=>m.moveId===id))&&wc(rows[0].explanation)>=8,`noveltyMap.${side}.${f}`,'Complete source-grounded novelty mapping required');}
}
check(out.sourceQuoteAudit?.length===2,'sourceQuoteAudit','Two representative quote audits required');check(out.noveltyMap?.length===8,'noveltyMap','Eight mapped extension items required');
console.log(JSON.stringify({schemaVersion:'1.0-comprehensive-staged-publication-validation',status:errors.length?'failed':'passed',debateNumber:number,debateId:record.debateId,output:fileRecord(outputPath),errors,metrics:{cards:cards.length,sections:c.sections.length,descriptionMean,critiques:cards.map(m=>({moveId:m.ledgerMoveId,path:`${m.path}.critique`,words:wc(m.critique),characters:m.critique.length}))}},null,2));
process.exitCode=errors.length?1:0;
