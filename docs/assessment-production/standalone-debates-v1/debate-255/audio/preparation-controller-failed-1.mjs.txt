import fs from 'node:fs';

 const seconds=clips.reduce((n,c)=>n+c.clip.durationSeconds,0)+clips.length*refs.reduce((n,r)=>n+r.clip.durationSeconds,0),maximum=round(clips.length*(16000*2.5+2000*10)/1e6);

 write(planPath,{schemaVersion:'1.1-bounded-attribution-and-lexical-audio-plan',status:'frozen-before-paid-calls',debateNumber:number,debateId:entry.debateId,inventory:record(inventoryPath),source:record(sourcePath),sourceDurationSeconds:duration,model:'gpt-4o-transcribe-diarize',responseFormat:'diarized_json',chunkingStrategy:'auto',language:'en',speakerReferences:refs,clips,estimatedSecondsIncludingReferences:seconds,cost:{estimatedUsd:round(seconds/60*.006),maximumUsd:maximum,maximumForDebateUsd:auth.cost.standingPaidTranscriptionMaximumUsd,priorKnownCostUsd:0,priorUncertainCostUsd:0,pricingUrl:'https://developers.openai.com/api/docs/pricing',modelPricingUrl:'https://developers.openai.com/api/docs/models/gpt-4o-transcribe-diarize',pricingRetrievedAt:new Date().toISOString(),inputPerMillionUsd:2.5,outputPerMillionUsd:10,planningInputTokensPerCall:16000,planningOutputTokensPerCall:2000,planningCaveat:'The time-based estimate is indicative; the conservative per-call planning reserve is not a server-enforced billing cap.  console.log(JSON.stringify({clips:clips.length,seconds,estimatedUsd:round(seconds/60*.006),maximumUsd:maximum}));
}else{
 const plan=read(planPath);assert.equal(plan.inventory.sha256,record(inventoryPath).sha256);assert.equal(plan.debateId,entry.debateId);assert.ok(plan.cost.maximumUsd<=auth.cost.standingPaidTranscriptionMaximumUsd);
 assert.deepEqual(plan.clips.map(c=>c.moveId),inventory.moves.filter(m=>m.attributionConfidence!=='high').map(m=>m.moveId));
 const reserve=round((plan.cost.planningInputTokensPerCall*plan.cost.inputPerMillionUsd+plan.cost.planningOutputTokensPerCall*plan.cost.outputPerMillionUsd)/1e6);
 if(phase==='disclose'){
  write(`${audioRoot}/cost-disclosure.json`,{status:'disclosed-before-paid-calls',debateNumber:number,plan:record(planPath),estimatedUsd:plan.cost.estimatedUsd,conservativeMaximumUsd:plan.cost.maximumUsd,standingMaximumUsd:auth.cost.standingPaidTranscriptionMaximumUsd,statement:'User was told the complete current-debate check estimate and conservative reserve before the first paid call; the installed skill authorizes cumulative paid transcription up to one dollar after disclosure.',disclosedAt:new Date().toISOString()});

  for(const r of [clip.clip,...plan.speakerReferences.map(r=>r.clip)])assert.equal(record(r.path).sha256,r.sha256);assert.ok(process.env.OPENAI_API_KEY,'OPENAI_API_KEY missing');
  const cli=plan.execution.cli;assert.deepEqual(record(cli.path),cli);

  const callArgs=['-c',wrapper,cli.path,clip.clip.path,'--model',plan.model,'--response-format',plan.responseFormat,'--chunking-strategy',plan.chunkingStrategy,'--language',plan.language,'--stdout',...plan.speakerReferences.flatMap(r=>['--known-speaker',r.name+'='+r.clip.path])];

  try{const body=execFileSync('/Library/Frameworks/Python.framework/Versions/3.13/bin/python3',callArgs,{encoding:'utf8',maxBuffer:10*1024*1024,timeout:250000});fs.writeFileSync(responsePath,body,{flag:'wx'});const data=JSON.parse(body);assert.ok(data.segments?.length);assert.equal(data.usage?.type,'tokens');const cost=round((data.usage.input_tokens*plan.cost.inputPerMillionUsd+data.usage.output_tokens*plan.cost.outputPerMillionUsd)/1e6);fs.mkdirSync(path.dirname(clip.responsePath),{recursive:true});fs.copyFileSync(responsePath,clip.responsePath,fs.constants.COPYFILE_EXCL);fs.writeFileSync(attemptPath,serializedJson({...attempt,status:'complete',completedAt:new Date().toISOString(),response:record(responsePath),usage:data.usage,directCostUsd:cost,maximumPotentialCostUsd:cost}));console.log(JSON.stringify({moveId:clip.moveId,segments:data.segments.length,directCostUsd:cost,cumulativeCostUsd:round(prior+cost)}));}catch(error){fs.writeFileSync(attemptPath,serializedJson({...attempt,status:'failed-or-transport-uncertain',finishedAt:new Date().toISOString(),error:String(error.message),stdout:typeof error.stdout==='string'?error.stdout:null,stderr:typeof error.stderr==='string'?error.stderr:null,noAutomaticRetry:true}));throw error;}
 }else if(phase==='finalize'){

 }else throw new Error('Unknown phase');
