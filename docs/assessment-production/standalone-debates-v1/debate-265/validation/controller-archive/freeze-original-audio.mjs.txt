import fs from 'node:fs';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
const registry=JSON.parse(fs.readFileSync('docs/assessment-production/standalone-debates-v1/registry.json','utf8'));
const record=registry.debates.find(x=>x.debateNumber==='265');
assert.equal(record.videoId,'Ui9C6N4HFvY');
const root='docs/assessment-production/standalone-debates-v1/debate-'+record.debateNumber;
const base='.assessment-cache/captions/'+record.videoId;
const info=JSON.parse(fs.readFileSync(base+'/original-audio.info.json','utf8'));
assert.equal(info.id,record.videoId);
assert.equal(info.language,'en-US');
assert.match(info.format_note,/original/);
const file=p=>{const b=fs.readFileSync(p);return {path:p,sha256:createHash('sha256').update(b).digest('hex'),bytes:b.length};};
const duration=Number(execFileSync('/opt/homebrew/bin/ffprobe',['-v','error','-show_entries','format=duration','-of','default=noprint_wrappers=1:nokey=1',base+'/original-audio.webm'],{encoding:'utf8'}).trim());
assert.ok(duration>6877 && duration<6878);
const out={schemaVersion:'1.0-original-audio-provenance',status:'acquired-and-hash-locked',debateNumber:record.debateNumber,debateId:record.debateId,videoId:record.videoId,canonicalUrl:'https://www.youtube.com/watch?v='+record.videoId,acquisition:{method:'Existing yt-dlp 2026.08.19, public original English audio, no paid acquisition',formatId:info.format_id,formatNote:info.format_note,language:info.language,durationSeconds:duration,metadataTitle:info.title,directCostUsd:0,downloadRetries:0,fragmentRetries:0,extractorRetries:0},files:{audio:file(base+'/original-audio.webm'),metadata:file(base+'/original-audio.info.json')},failedEarlierAttempt:{method:'Direct public Android player audio URL',httpStatus:403,paidService:false,directCostUsd:0,usableAudioWritten:false,disposition:'No retry of that transport; existing downloader independently selected original English audio.'},captionLockUnchanged:file(root+'/source/source-lock.json'),scope:'Original audio supplements the fixed complete captions for required wording and speaker checks; acquisition alone is not a claim that the audio has been listened to or transcribed.'};
fs.writeFileSync(root+'/source/original-audio-provenance.json',JSON.stringify(out,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify(out,null,2));
