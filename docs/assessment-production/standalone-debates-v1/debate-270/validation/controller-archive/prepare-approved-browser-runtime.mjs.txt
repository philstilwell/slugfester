import fs from 'node:fs';
import assert from 'node:assert/strict';
import {avatarsForSpeakerText} from '../src/data/interlocutors.js';
import {interlocutorPath} from '../src/seo.js';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1],port=Number(args[args.indexOf('--port')+1]);assert(args.includes('--debate')&&args.includes('--port'));assert.match(n,/^\d{3,}$/);assert(Number.isInteger(port)&&port>1024&&port<65536);
const read=p=>JSON.parse(fs.readFileSync(p)),r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
const c=read(r.root+'/publication/output.json').candidate,people=['pro','con'].map(side=>{const people=avatarsForSpeakerText(c.sides[side].speaker);assert.equal(people.length,1);return people[0];});
const expected={number:n,origin:'http://127.0.0.1:'+port,path:'/debate/'+c.id+'/',topicPath:'/topics/'+c.topicCategory+'/',source:c.youtubeUrl,scores:c.score,speakers:people.map(p=>p.name),profiles:people.map(interlocutorPath),byline:'Assessments made by '+c.assessmentModel+'. — Rubric: '+c.assessmentRubric+'.'};
const body=async(page)=>{
 const errors=[],failures=[],responses=[],pages=[],check=(v,m)=>{if(!v)throw Error(m);};
 const onError=e=>errors.push(String(e)),onConsole=m=>{if(m.type()==='error')errors.push(m.text());},onFail=r=>failures.push({url:r.url(),error:r.failure()?.errorText}),onResponse=r=>{if(r.status()>=400)responses.push({url:r.url(),status:r.status()});};
 page.on('pageerror',onError);page.on('console',onConsole);page.on('requestfailed',onFail);page.on('response',onResponse);
 const routes=[expected.path,'/','/search/?q='+encodeURIComponent(expected.speakers[0]),'/topics/',expected.topicPath,...expected.profiles,'/rankings/?compare-a='+encodeURIComponent(expected.speakers[0])+'&compare-b='+encodeURIComponent(expected.speakers[1])];
 try{for(const width of [1440,390]){await page.setViewportSize({width,height:width===1440?1000:844});for(const route of routes){await page.goto(expected.origin+route,{waitUntil:'networkidle'});await page.locator('main h1').first().waitFor();const overflow=await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth);check(!overflow,'Overflow '+route+' '+width);let biographyPlacement=null;
  if(route===expected.path){const scoreText=await page.getByRole('complementary',{name:'Debate score summary'}).innerText();for(const side of ['pro','con'])check(new RegExp('\\b'+expected.scores[side]+'\\b').test(scoreText),'Score '+side);check((await page.locator('body').innerText()).replace(/\s+/g,' ').includes(expected.byline),'Byline');check(await page.getByRole('link',{name:'Open YouTube source',exact:true}).getAttribute('href')===expected.source,'Source URL');}
  else if(route.startsWith('/rankings/')){const t=await page.locator('body').innerText();check(expected.speakers.every(s=>t.includes(s)),'Rankings participants');}
  else if(route==='/topics/'){check(await page.locator('a[href="'+expected.topicPath+'"]').count()>0,'Topic index links current topic');}
  else{check(await page.locator('a[href="'+expected.path+'"]').count()>0,'Missing new-debate link '+route);}
  if(expected.profiles.includes(route)){const b=await page.locator('.profile-biography').boundingBox(),g=await page.locator('.profile-score-histogram').boundingBox();check(b&&g,'Biography and graph');biographyPlacement=width===1440?'left':'above';check(width===1440?b.x+b.width<=g.x+1:b.y+b.height<=g.y+1,'Biography placement '+width);await page.locator('.profile-bio-chart').scrollIntoViewIfNeeded();await page.screenshot({path:'output/playwright/debate-'+expected.number+'/'+(width===1440?'desktop':'mobile')+'-'+route.split('/')[2]+'-profile.png'});}
  pages.push({route,width,title:await page.title(),overflow,biographyPlacement});
 }}check(!errors.length&&!failures.length&&!responses.length,JSON.stringify({errors,failures,responses}));return {status:'passed-final-browser-runtime-replay',pages,errors,failedRequests:failures,failedResponses:responses};}
 finally{page.off('pageerror',onError);page.off('console',onConsole);page.off('requestfailed',onFail);page.off('response',onResponse);}
};
const out='output/playwright/debate-'+n;fs.mkdirSync(out,{recursive:true});fs.writeFileSync(out+'/runtime-replay.js','async (page) => { const expected='+JSON.stringify(expected)+'; return ('+body.toString()+')(page); }\n',{flag:'wx'});
console.log(JSON.stringify({status:'prepared-current-runtime-replay',expected,output:out+'/runtime-replay.js'}));
