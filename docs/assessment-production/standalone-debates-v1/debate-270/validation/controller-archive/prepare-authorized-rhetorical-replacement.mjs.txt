import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1];
assert(args.includes('--debate'));assert.match(n,/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p)),hash=b=>createHash('sha256').update(b).digest('hex');
const rec=p=>{const b=fs.readFileSync(p);return {path:p,sha256:hash(b),bytes:b.length};};
const write=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n',{flag:'wx'});
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
const base=r.root+'/publication/rhetorical-tag-review-1',dir=base+'/pass-a-replacement-1';
const old=read(base+'/pass-a/execution-plan.json'),pause=read(r.root+'/publication/resume-after-rhetorical-quality-gate.json');
assert.equal(pause.blockingFailure.phase,'rhetorical-pass-a');assert.equal(pause.blockingFailure.attemptConsumed,1);
assert.equal(rec(base+'/pass-b/output.json').sha256,pause.retainedPassB.sha256);
const rejection=read(base+'/pass-a/controller-quality-rejection.json');assert.equal(rec(base+'/pass-a/output.json').sha256,rejection.output.sha256);
for(const x of old.allowedInputs)assert.equal(rec(x.path).sha256,x.sha256);
assert.equal(rec(r.root+'/publication/output.json').sha256,rec(base+'/preserved-prior-output.json').sha256);
fs.mkdirSync(dir);
const authority={schemaVersion:'1.0-exceptional-rhetorical-replacement-authorization',status:'user-approved-before-replacement',debateNumber:n,debateId:r.debateId,createdAt:new Date().toISOString(),userReply:'Yes.',approvedQuestion:'Replace only the failed rhetorical pass-a with one fresh isolated reviewer, retaining valid pass-b and source evidence; also complete the scheduled measured performance review with unchanged effective download limits.',supersedesPause:rec(r.root+'/publication/resume-after-rhetorical-quality-gate.json'),preservedFailedOutput:rec(base+'/pass-a/output.json'),preservedValidPassB:rec(base+'/pass-b/output.json'),preservedPublication:rec(r.root+'/publication/output.json'),failedPassAttempts:1,exceptionalReplacementAttempts:1,replacementRetries:0,additionalDirectCostUsd:0,judgmentChanges:0,scoreChanges:0,proseChanges:0,reviewerInputsExcludePriorReviews:true,selectedReviewDirectories:{'pass-a':'pass-a-replacement-1','pass-b':'pass-b'}};
write(base+'/replacement-authorization.json',authority);
write(dir+'/execution-plan.json',{...old,outputPath:path.resolve(dir+'/output.json'),readingAuditPath:path.resolve(dir+'/reading-audit.json'),firstSaveRequiresControllerReviewOfUnsavedCompleteReasons:true,sourceSpecificReasonRequirement:'Every move and blunder explanation must name its actual claim or inference and explain the applicable definition boundary; a generic statement that all definitions were considered is not sufficient. Produce original review reasoning, not copied publication critiques.',exceptionalAttempt:1});
fs.mkdirSync(r.root+'/validation',{recursive:true});
write(r.root+'/validation/performance-review-authorization.json',{schemaVersion:'1.0-measured-performance-review-authorization',status:'user-approved-before-measurement',debateNumber:n,debateId:r.debateId,createdAt:new Date().toISOString(),userReply:'Yes.',scope:'Complete the scheduled performance review and update the catalogue baseline only if all measured routes pass existing effective decoded data-download limits. Preserve every effective limit, per-debate growth allowance and review interval. No scoring change.',additionalDirectCostUsd:0});
console.log(JSON.stringify({status:'approved-replacement-plan-frozen',planPath:dir+'/execution-plan.json',preservedPublication:authority.preservedPublication,retainedPassB:authority.preservedValidPassB}));
