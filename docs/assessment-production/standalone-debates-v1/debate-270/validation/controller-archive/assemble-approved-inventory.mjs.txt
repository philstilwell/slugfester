import fs from 'node:fs';
import assert from 'node:assert/strict';
import path from 'node:path';
import {validateEditorScopedSource} from '../scripts/lib/assessment-editor-scoped-source.mjs';
import {fileRecord as repositoryFileRecord,validateStandaloneInventory} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
const args=process.argv.slice(2),di=args.indexOf('--debate');assert(di>=0);const n=args[di+1];assert.match(n,/^\d{3,}$/);
const fileRecord=p=>repositoryFileRecord(path.isAbsolute(p)?path.relative(process.cwd(),p):p);
const read=p=>JSON.parse(fs.readFileSync(p)),r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
const root=r.root,a=read(root+'/authorization.json'),source=read(root+'/source/source-lock.json'),packet=read(root+'/inventory/packet.json'),manifest=read(root+'/manifest.json');
const draftPath=args.find(x=>x.startsWith('--draft='))?.slice(8)||packet.outputPath,d=read(draftPath),events=read(manifest.sourceLocks.events.path);
assert.equal(d.motion,a.identity.motion);assert.equal(packet.identity.debateId,r.debateId);
const inv=structuredClone(d);Object.assign(inv,{schemaVersion:'1.2-standalone-score-blind-inventory',protocolId:a.protocolId,status:'complete-and-frozen',debateNumber:n,debateId:r.debateId,motion:a.identity.motion,assessmentModel:a.execution.recordedDisplayModel});
for(const m of inv.moves){
 const startEvent=m.startEvent??m.sourceSpan?.startEvent,endEvent=m.endEvent??m.sourceSpan?.endEvent;
 assert(Number.isInteger(startEvent)&&Number.isInteger(endEvent)&&endEvent>=startEvent&&endEvent<events.length,m.moveId+': invalid span');
 const sourceSpan={startEvent,endEvent,startMs:events[startEvent].startMs,endMs:events[endEvent].startMs+events[endEvent].durationMs,excerpt:events.slice(startEvent,endEvent+1).map(e=>e.text.replace(/\s+/g,' ').trim()).join(' ').replace(/\s+/g,' ').trim()};
 assert(sourceSpan.startMs>=source.participants.assessedDebateWindowMs.start&&sourceSpan.endMs<=source.participants.assessedDebateWindowMs.end,m.moveId+': outside substantive source envelope');
 if(m.sourceSpan?.excerpt)assert.equal(m.sourceSpan.excerpt,sourceSpan.excerpt,m.moveId+': excerpt mismatch');
 m.sourceSpan=sourceSpan;delete m.startEvent;delete m.endEvent;
 assert(sourceSpan.excerpt.length<=manifest.inventoryControls.sourceExcerptTargetMaximumCharacters||String(m.sourceSpanSelectionRationale||'').length>=40,m.moveId+': long span without specific rationale');
}
inv.moves.sort((x,y)=>x.sourceSpan.startEvent-y.sourceSpan.startEvent);
const counts=side=>inv.moves.filter(m=>m.side===side).length;
const sections=inv.sections.map(s=>{const pro=inv.moves.filter(m=>m.sectionId===s.sectionId&&m.side==='pro').length,con=inv.moves.filter(m=>m.sectionId===s.sectionId&&m.side==='con').length;return{sectionId:s.sectionId,pro,con,absoluteDifference:Math.abs(pro-con)};});
const balance={thresholds:{totalAbsoluteDifference:3,sectionAbsoluteDifference:2},totals:{pro:counts('pro'),con:counts('con'),absoluteDifference:Math.abs(counts('pro')-counts('con'))},sections,rationaleRequired:Math.abs(counts('pro')-counts('con'))>=3||sections.some(s=>s.absoluteDifference>=2),rationale:d.selectionBalanceAudit?.rationale||''};
assert(d.selectionBalanceAudit);for(const k of ['thresholds','totals','sections','rationaleRequired'])assert.deepEqual(d.selectionBalanceAudit[k],balance[k],'Worker balance discrepancy: '+k);
inv.selectionBalanceAudit=balance;
const capacities=sections.map(s=>{const proposed=d.publicationCapacityAudit?.sections?.find(x=>x.sectionId===s.sectionId),fourthRowRequired=s.pro===4||s.con===4;return{sectionId:s.sectionId,pro:s.pro,con:s.con,withinCapacity:s.pro<=4&&s.con<=4,fourthRowRequired,fourthRowAuthorized:!fourthRowRequired||(proposed?.fourthRowAuthorized===true&&String(proposed?.rationale||'').length>=40),rationale:proposed?.rationale||''};});
assert(d.publicationCapacityAudit);for(const s of capacities){const proposed=d.publicationCapacityAudit.sections.find(x=>x.sectionId===s.sectionId);for(const k of ['pro','con','withinCapacity','fourthRowRequired','fourthRowAuthorized'])assert.equal(proposed?.[k],s[k],'Worker capacity discrepancy: '+s.sectionId+':'+k);}
inv.publicationCapacityAudit={maximumSelectedMovesPerSidePerSection:4,sections:capacities,allSectionsWithinCapacity:capacities.every(x=>x.withinCapacity),allFourthRowsPreauthorized:capacities.every(x=>x.fourthRowAuthorized)};
for(const k of ['allSectionsWithinCapacity','allFourthRowsPreauthorized'])assert.equal(d.publicationCapacityAudit[k],inv.publicationCapacityAudit[k]);
inv.assessedDebateWindowMs=source.participants.assessedDebateWindowMs;
inv.audit={...d.audit,calculatedTotalsAbsent:true,completeTranscriptReviewed:true,allSpansSourceExact:true,originalRecordingReviewedByInventoryWorker:false,fullIndexedTranscriptReviewedByInventoryWorker:true};
if(a.identity.editorApprovedScope){inv.editorApprovedScope=a.identity.editorApprovedScope;validateEditorScopedSource({authorization:a,sourceLock:source,inventory:inv});}
const validation=validateStandaloneInventory(inv,events);
const review={schemaVersion:'1.0-current-source-inventory-assembly',status:'passed-mechanical-review-source-review-pending',createdAt:new Date().toISOString(),debateNumber:n,debateId:r.debateId,draft:fileRecord(draftPath),sourceEvents:manifest.sourceLocks.events,validation,assembly:'Worker-authored semantic selection preserved. Exact source endpoints/text, chronology, total and section-side counts independently derived by controller. No ratings or scores.',counts:balance,capacity:inv.publicationCapacityAudit,moveSpans:inv.moves.map(m=>({moveId:m.moveId,side:m.side,sectionId:m.sectionId,startEvent:m.sourceSpan.startEvent,endEvent:m.sourceSpan.endEvent,characters:m.sourceSpan.excerpt.length,attributionConfidence:m.attributionConfidence})),directIncrementalCostUsd:0};
if(args.includes('--preserve'))for(const [p,v] of [[root+'/inventory/pre-audio-inventory.json',inv],[root+'/inventory/pre-audio-mechanical-review.json',review]])fs.writeFileSync(p,JSON.stringify(v,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({validation,counts:balance,capacity:inv.publicationCapacityAudit,moveSpans:review.moveSpans,preserved:args.includes('--preserve')},null,2));
