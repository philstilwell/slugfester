import fs from 'node:fs';
import assert from 'node:assert/strict';
import {debates} from '../src/data/debates.js';
import {debateAnalytics} from '../src/data/debate-analytics.js';
import {fileRecord} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1],port=Number(args[args.indexOf('--port')+1]);
assert(args.includes('--debate')&&args.includes('--port'));assert.match(n,/^\d{3,}$/);assert(Number.isInteger(port)&&port>1024&&port<65536);
const read=p=>JSON.parse(fs.readFileSync(p)),r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
assert(debates.some(d=>d.id===r.debateId));assert.deepEqual(Object.keys(debateAnalytics).sort(),debates.map(d=>d.id).sort());
const rows=debates.map(d=>{const scores=d.sections.flatMap(s=>[s.score.pro,s.score.con].filter(Number.isFinite));assert.deepEqual(debateAnalytics[d.id].sectionScores,scores,'Analytics mismatch '+d.id);return{id:d.id,number:d.number,scores};});
const scores=rows.flatMap(r=>r.scores),low=Math.min(...scores),high=Math.max(...scores),buckets=[];
for(let start=Math.floor(low/2)*2;start<=Math.floor(high/2)*2;start+=2)buckets.push({start,end:Math.min(start+1,100),count:scores.filter(s=>s>=start&&s<start+2).length});
const expected={number:n,origin:'http://127.0.0.1:'+port,total:scores.length,range:low+'–'+high,buckets,axisMaximum:Math.ceil(Math.max(...buckets.map(b=>b.count))*1.1/20)*20,newDebate:rows.find(x=>x.id===r.debateId)};
assert.equal(expected.newDebate.scores.length,read(r.root+'/publication/output.json').candidate.sections.length*2);
const evidence={status:'passed-published-catalogue-to-generated-analytics-reconciliation',debateNumber:n,debateId:r.debateId,at:new Date().toISOString(),debates:rows.length,sectionSideScores:expected.total,observedRange:expected.range,buckets,newDebateContribution:expected.newDebate,source:fileRecord('src/data/debates.js'),analytics:fileRecord('src/data/debate-analytics.js'),scoreCalculatorInvoked:false};
fs.mkdirSync(r.root+'/rendering',{recursive:true});fs.writeFileSync(r.root+'/rendering/backend-catalogue-audit.json',JSON.stringify(evidence,null,2)+'\n',{flag:'wx'});
const body=async(page)=>{
 const check=(v,m)=>{if(!v)throw Error(m);};
 await page.goto(expected.origin+'/backend/',{waitUntil:'networkidle'});
 const chart=page.locator('#rubric-quality-check');await chart.waitFor();
 const summary=(await chart.locator('.section-score-distribution-summary dd').allTextContents()).map(x=>x.trim());
 check(JSON.stringify(summary)===JSON.stringify([expected.total.toLocaleString('en-US'),expected.range,'2 points']),'Backend summary');
 const labels=await chart.locator('.section-score-bucket').evaluateAll(es=>es.map(e=>e.getAttribute('aria-label')));
 const required=expected.buckets.map(b=>b.start+'–'+b.end+' percent: '+b.count.toLocaleString('en-US')+' section-side score'+(b.count===1?'':'s'));
 check(JSON.stringify(labels)===JSON.stringify(required),'Every backend bucket');
 check((await chart.locator('.section-score-y-axis span').first().innerText())===expected.axisMaximum.toLocaleString('en-US'),'Backend vertical scale');
 check(!(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth)),'Backend overflow');
 await chart.scrollIntoViewIfNeeded();await page.screenshot({path:'output/playwright/debate-'+expected.number+'/backend-distribution.png'});
 return{status:'passed-rendered-backend-catalogue-reconciliation',total:expected.total,range:expected.range,buckets:expected.buckets,newDebateContribution:expected.newDebate,horizontalOverflow:false};
};
const out='output/playwright/debate-'+n;fs.mkdirSync(out,{recursive:true});fs.writeFileSync(out+'/check-backend.js','async(page)=>{const expected='+JSON.stringify(expected)+';return ('+body.toString()+')(page);}\n',{flag:'wx'});
console.log(JSON.stringify(evidence,null,2));
