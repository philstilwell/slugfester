import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {fileRecord as repositoryFileRecord} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
const args=process.argv.slice(2), n=args[args.indexOf('--debate')+1];
assert(args.includes('--debate')); assert.match(n,/^\d{3,}$/);
const read=p=>JSON.parse(fs.readFileSync(p)), rec=p=>repositoryFileRecord(path.isAbsolute(p)?path.relative(process.cwd(),p):p);
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n); assert(r);
const root=r.root,a=read(root+'/authorization.json'),manifest=read(root+'/manifest.json'),inv=read(root+'/inventory/inventory.json'),audio=read(root+'/audio/audio-verification.json');
assert.equal(inv.debateId,r.debateId); assert.equal(inv.motion,a.identity.motion); assert.equal(audio.status,'complete'); assert.equal(audio.audit.unresolvedAttributionChecks,0);
const source=manifest.sourceLocks.indexedTranscript.path; assert.equal(rec(source).sha256,manifest.sourceLocks.indexedTranscript.sha256);
const inputs=[rec(root+'/judgments/judgment-packet.json'),rec(root+'/inventory/inventory.json'),rec(source),rec(root+'/audio/score-blind-supplement.json'),rec('docs/assessment-production-workflow.md'),rec('docs/reassessment-rubric-v2.1.md'),{...rec('scripts/lib/assessment-production-standalone-debate-v1.mjs'),allowedLines:[{start:636,end:795}]}];
const common={schemaVersion:'1.0-frozen-isolated-primary-instructions',debateNumber:n,debateId:r.debateId,model:{slug:a.execution.assessmentModel,displayLabel:a.execution.recordedDisplayModel,reasoningEffort:a.execution.reasoningEffort,authentication:a.execution.authentication},allowedInputs:inputs,sourceEventCount:fs.readFileSync(source,'utf8').trim().split('\n').length,inventoryMoveCount:inv.moves.length,identity:a.identity,requirements:[
'Read every allowed input literally and completely, except the validator restricted to its listed lines. Transcript chunks at most 300 lines; inventory and packet in untruncated chunks. Send a reading checkpoint and wait for authenticated controller release before rating. Hashing or scripted extraction does not substitute for reading.',
'Fresh isolated primary assessment. Do not read the other pass, other debates, existing assessments, totals, winners, publication prose, Git history, biographies, external research, credentials or paid APIs. Use env -u OPENAI_API_KEY -u ANTHROPIC_API_KEY for shell calls. No additional agents.',
'Rate each locked move exactly once in inventory order on all six dimensions. Independently author values and dimension-specific source-specific rationales of at least forty characters. Do not use a formula, repeated rationale or blanket rating template. Apply rubric band anchors to the actual inference.',
'Assess only the explicitly approved formal rounds and closing statements. The entire audience Q&A and both speakers\' answers are excluded. Preserve absolute timestamps; the transcript gap is intentional. Neither side receives credit or penalty from excluded material, including alleged unanswered objections. A closing must stand on its own stated premises or retained formal-round evidence. Moderator logistics are not arguments.',
'The authorized question and side mapping are in identity and must remain byte-identical. Distinguish an evidential challenge from a positive denial or moral claim actually adopted. Criticizing the affirmative case does not by itself oblige the other side to prove an entire alternative worldview. Do not score a worldview as true or false.',
'Audio-derived clarifications preserve raw transcription quotations. Never penalize recognition errors, silently normalize disputed verse references or Greek words, or claim to have listened to audio. Treat promptly corrected slips with their corrections. Assessment confidence is distinct from source attribution confidence.',
'Do not alter inventory, sections, weights, importance, burden bridges, evidence or reply links. No calculated move, section or overall scores, winner or rhetorical labels. Supply only dimension ratings and rationales plus the two independently evaluated burden-completion adjustments.',
'The exact nine-field eligibility object is mandatory for each burden adjustment. A nonzero value needs a distinct debate-wide consequence not captured in any selected move; double-counting forces zero. Do not use adjustments as impression bonuses or penalties.',
'One first submission, no automatic retries or replacements. Check the schema and full move order in memory before the first write. Never invoke a calculator. Use apply_patch only at the declared absolute output and reading-audit paths. Do not overwrite a submitted artifact. No invented deadline; continue through context compaction if necessary.'
],directIncrementalCostUsd:0,attempts:1,retries:0};
const passes=[];
for(const pass of ['pass-a','pass-b']){
 const dir=root+'/judgments/'+pass; fs.mkdirSync(dir,{recursive:true});
 const instructions={...common,pass,outputPath:path.resolve(dir+'/output.json'),readingAuditPath:path.resolve(dir+'/reading-audit.json')},p=dir+'/execution-instructions.json';
 fs.writeFileSync(p,JSON.stringify(instructions,null,2)+'\n',{flag:'wx'});
 passes.push({pass,instructions:rec(p),outputPath:instructions.outputPath,readingAuditPath:instructions.readingAuditPath,attempts:1,retries:0});
}
const plan={schemaVersion:'1.0-frozen-primary-execution-plan',status:'frozen-before-primary-execution',createdAt:new Date().toISOString(),debateNumber:n,debateId:r.debateId,execution:{model:a.execution.assessmentModel,displayModel:a.execution.recordedDisplayModel,reasoningEffort:a.execution.reasoningEffort,authentication:a.execution.authentication,freshContextPerPass:true,forkTurns:'none',inventorySha256:rec(root+'/inventory/inventory.json').sha256,isolation:'Protocol-restricted fresh built-in contexts and disjoint output paths; not an operating-system filesystem sandbox.'},allowedInputs:inputs,passes,directIncrementalCostUsd:0};
fs.writeFileSync(root+'/judgments/execution-plan.json',JSON.stringify(plan,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({debateNumber:n,moves:inv.moves.length,passes},null,2));
