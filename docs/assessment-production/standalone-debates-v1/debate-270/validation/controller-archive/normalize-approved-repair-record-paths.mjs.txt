import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
const args=process.argv.slice(2), n=args[args.indexOf('--debate')+1];
assert(args.includes('--debate')&&/^\d{3,}$/.test(n));
const read=p=>JSON.parse(fs.readFileSync(p));
const rec=p=>{const b=fs.readFileSync(p);return {path:p,sha256:createHash('sha256').update(b).digest('hex'),bytes:b.length};};
const put=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n',{flag:'wx'});
const rp='docs/assessment-production/standalone-debates-v1/registry.json',registry=read(rp),r=registry.debates.find(x=>x.debateNumber===n);
assert(r&&r.status==='published-and-frozen');
const base=r.root+'/publication/field-repair-1',dir=r.root+'/validation/repair-record-paths';
assert(!fs.existsSync(dir));fs.mkdirSync(dir);
const ep=r.contentParity.executionPath,cp=r.contentParity.correctionPath,lp=r.productionLedger.path;
assert.equal(ep,base+'/execution.json');
const execution=read(ep),correction=read(cp),adapter=read(lp),changes=[];
assert.equal(execution.debateId,r.debateId);
const originalRecords=[ep,cp,lp,rp].map(rec);
for(const [i,p] of [ep,cp,lp,rp].entries())fs.copyFileSync(p,dir+'/original-'+i+'.json',fs.constants.COPYFILE_EXCL);
function portable(record){
  if(!path.isAbsolute(record.path))return record;
  assert(record.path.startsWith(process.cwd()+'/'));
  const relative=path.relative(process.cwd(),record.path);assert(relative.startsWith(r.root+'/'));
  assert.equal(rec(relative).sha256,record.sha256);assert.equal(rec(relative).bytes,record.bytes);
  changes.push({before:record.path,after:relative,sha256:record.sha256});
  return {...record,path:relative};
}
execution.contexts=execution.contexts.map(portable);
execution.shards=execution.shards.map(s=>({...s,input:portable(s.input),output:portable(s.output)}));
assert.equal(execution.shards.length,correction.writableShards.length);
const newEp=base+'/execution-portable.json',newCp=r.root+'/publication/correction-content-parity-portable.json';
put(newEp,execution);
correction.writableShards=correction.writableShards.map(s=>({...s,input:portable(s.input),output:portable(s.output)}));
correction.evidence.execution=rec(newEp);put(newCp,correction);
const evidence=Object.values(adapter).find(v=>v&&typeof v==='object'&&v.publicationContentParityExecution);
assert(evidence);evidence.publicationContentParityExecution=rec(newEp);evidence.publicationContentParityCorrection=rec(newCp);
const beforeAdapter=read(lp);const expected=structuredClone(beforeAdapter);
const expectedEvidence=Object.values(expected).find(v=>v&&typeof v==='object'&&v.publicationContentParityExecution);
expectedEvidence.publicationContentParityExecution=rec(newEp);expectedEvidence.publicationContentParityCorrection=rec(newCp);
assert.deepEqual(adapter,expected);
fs.writeFileSync(lp,JSON.stringify(adapter,null,2)+'\n');
r.contentParity.executionPath=newEp;r.contentParity.correctionPath=newCp;r.productionLedger=rec(lp);
fs.writeFileSync(rp,JSON.stringify(registry,null,2)+'\n');
put(dir+'/audit.json',{status:'passed-derived-file-record-path-normalization',debateNumber:n,debateId:r.debateId,reason:'Repository replay joins artifact file-record paths to its root; absolute paths repeated the root. New portable aggregate records preserve all original records and underlying worker outputs.',originalRecords,preserved:originalRecords.map((_,i)=>rec(dir+'/original-'+i+'.json')),activeExecution:rec(newEp),activeCorrection:rec(newCp),activeAdapter:rec(lp),activeRegistry:rec(rp),pathChanges:changes,workerOutputChanges:0,sourceChanges:0,judgmentChanges:0,scoreChanges:0,publicationChanges:0,modelAttempts:0,directIncrementalCostUsd:0});
console.log(JSON.stringify({status:'portable-derived-records-registered',pathChanges:changes.length}));
