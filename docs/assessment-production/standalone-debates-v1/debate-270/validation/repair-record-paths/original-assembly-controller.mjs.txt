import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {checkRepair} from './check-publication-field-repair.mjs';import {checkPublication} from './check-approved-publication.mjs';
const a=process.argv.slice(2),n=a[a.indexOf('--debate')+1],mode=a[a.indexOf('--mode')+1];assert(a.includes('--debate')&&a.includes('--mode')&&/^\d{3,}$/.test(n));assert(['assemble','register'].includes(mode));
const read=p=>JSON.parse(fs.readFileSync(p)),rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:createHash('sha256').update(b).digest('hex'),bytes:b.length};},put=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n',{flag:'wx'});
const rp='docs/assessment-production/standalone-debates-v1/registry.json',registry=read(rp),r=registry.debates.find(x=>x.debateNumber===n);assert(r);const pub=r.root+'/publication',base=pub+'/field-repair-1',plan=read(base+'/plan.json');
if(mode==='assemble'){
 assert.equal(rec(pub+'/output.json').sha256,plan.preservedPriorOutput.sha256);const o=read(pub+'/output.json'),changes=[],shards=[],contexts=[];
 for(const g of plan.groups){assert.equal(rec(g.path).sha256,g.sha256);const p=read(g.path),result=read(p.outputPath),exec=read(p.outputPath.replace('/output.json','/execution.json'));assert.equal(exec.status,'passed-authenticated-first-field-repair');assert.equal(exec.output.sha256,rec(p.outputPath).sha256);checkRepair(result,read(p.packetPath));contexts.push(rec(p.outputPath.replace('/output.json','/execution.json')));
 for(const shard of result.shards){const declared=p.shards.find(x=>x.shardId===shard.shardId);assert(declared&&shard.fields.length<=2);for(const f of shard.fields){assert(!changes.includes(f.path));const keys=f.path.replace(/\[(\d+)\]/g,'.$1').split('.');let target=o;for(const k of keys.slice(0,-1)){assert(k in target);target=target[k];}assert.notEqual(target[keys.at(-1)],f.text);target[keys.at(-1)]=f.text;changes.push(f.path);}shards.push({shardId:shard.shardId,writableFieldCount:shard.fields.length,fields:shard.fields.map(f=>f.path),input:rec(p.packetPath),output:rec(p.outputPath),attempts:1,retries:0});}}
 const notes=read(base+'/controller-editorial-acceptance.json');assert.equal(notes.status,'passed-direct-review-of-all-repaired-leaves');assert.deepEqual(notes.outputs,contexts.map(c=>rec(c.path.replace('/execution.json','/output.json'))));
 const validation=checkPublication(o,read(pub+'/input-bundle.json'));put(base+'/accepted-repaired-output.json',o);
 put(base+'/execution.json',{status:'passed-isolated-card-depth-repair',debateNumber:n,debateId:r.debateId,model:{label:read(pub+'/execution-plan.json').model.displayLabel,slug:read(pub+'/execution-plan.json').model.slug,reasoningEffort:read(pub+'/execution-plan.json').model.reasoningEffort},contexts,shards,audit:{judgmentChanges:0,scoreChanges:0,moveChanges:0,tagChanges:0,maximumWritableFieldsPerShard:2},directIncrementalCostUsd:0});
 put(base+'/assembly-audit.json',{status:'applied-once-and-frozen',debateNumber:n,debateId:r.debateId,before:rec(base+'/preserved-prior-output.json'),after:rec(base+'/accepted-repaired-output.json'),execution:rec(base+'/execution.json'),changedLeafFields:changes,validation,judgmentChanges:0,scoreChanges:0,moveChanges:0,tagChanges:0});
 fs.copyFileSync(base+'/accepted-repaired-output.json',pub+'/output.json');console.log(JSON.stringify({status:'editorial-repairs-assembled',shards:shards.length,fields:changes.length,validation}));
}else{
 const assembly=read(base+'/assembly-audit.json'),correction=pub+'/correction-content-parity.json';assert.equal(rec(base+'/accepted-repaired-output.json').sha256,assembly.after.sha256);const after=rec(base+'/accepted-repaired-output.json');
 put(correction,{schemaVersion:'1.0-standalone-content-parity-correction',status:'applied-once-and-frozen',debateNumber:n,debateId:r.debateId,preservedPriorOutput:rec(base+'/preserved-prior-output.json'),evidence:{before:rec(base+'/preserved-prior-output.json'),after,contentParityAudit:rec(pub+'/content-parity-audit.json'),execution:rec(base+'/execution.json')},writableShards:read(base+'/execution.json').shards,audit:{judgmentChanges:0,scoreChanges:0,moveChanges:0,tagChanges:0,maximumWritableFieldsPerShard:2,changedLeafFields:assembly.changedLeafFields.length},directIncrementalCostUsd:0});
 r.contentParity={auditPath:pub+'/content-parity-audit.json',correctionPath:correction,preservedOutputPath:base+'/preserved-prior-output.json',executionPath:base+'/execution.json'};fs.writeFileSync(rp,JSON.stringify(registry,null,2)+'\n');console.log('Editorial repair chain registered.');
}
