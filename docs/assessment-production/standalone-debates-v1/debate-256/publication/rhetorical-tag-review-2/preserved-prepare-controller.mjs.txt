import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {fileRecord,serializedJson} from './lib/assessment-production-standalone-debate-v1.mjs';
const args=process.argv.slice(2),get=k=>args[args.indexOf(k)+1],number=get('--debate');
assert.match(number??'',/^\d+$/);
const read=p=>JSON.parse(fs.readFileSync(p));
const entry=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(d=>d.debateNumber===number);assert(entry);
const root=entry.root,prior=`${root}/publication/rhetorical-tag-review-1`,next=`${root}/publication/rhetorical-tag-review-2`;
const write=(p,x)=>{assert(!fs.existsSync(p));fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,serializedJson(x),{flag:'wx'});};
const copy=(a,b)=>{assert(!fs.existsSync(b));fs.mkdirSync(path.dirname(b),{recursive:true});fs.copyFileSync(a,b,fs.constants.COPYFILE_EXCL);assert.equal(fileRecord(a).sha256,fileRecord(b).sha256);};
assert.equal(read(`${prior}/pass-b-semantic-validation.json`).status,'failed-source-specific-context-and-rationale-contract');
assert.equal(read(`${prior}/union-semantic-quarantine.json`).status,'quarantined-before-adjudication');
assert.equal(read(`${prior}/pass-a-reading-gate.json`).status,'passed');
const oldPlan=read(`${prior}/execution-plan.json`),oldPacket=read(`${prior}/packet.json`),model=oldPlan.model;
for(const r of oldPlan.allowedInputs)assert.deepEqual(fileRecord(r.path),r);
const authorization={status:'explicitly-user-authorized-one-fresh-pass-b-replacement',debateNumber:number,debateId:entry.debateId,authorizedAt:new Date().toISOString(),userReply:'yes.',approvedQuestion:'May I replace only that failed review with one fresh isolated reviewer, at $0 additional cost, preserving all valid work?',scope:'Replace only the failed second fallacy-and-bias review. Retain the valid first review, complete source, audio, inventory, primary judgments, score adjudication, single calculated scores, and accepted publication prose.',freshReplacementContexts:1,attempts:1,automaticRetries:0,directIncrementalCostEstimateUsd:0,preservedFirstReview:fileRecord(`${prior}/pass-a-output.json`),preservedFailedSecondReview:fileRecord(`${prior}/pass-b-output.json`),failure:fileRecord(`${prior}/pass-b-semantic-validation.json`),quarantinedUnion:fileRecord(`${prior}/union-semantic-quarantine.json`),frozenPublication:fileRecord(`${root}/publication/output.json`),frozenScores:fileRecord(`${root}/score-pass/output.json`)};
write(`${next}/authorization.json`,authorization);
for(const name of ['packet.json','reference-definitions.json','pass-a-output.json','pass-a-reading-receipt.json','pass-a-reading-gate.json'])copy(`${prior}/${name}`,`${next}/${name}`);
const guide=read(`${prior}/reading-guidance.json`);
write(`${next}/reading-guidance.json`,{...guide,status:'frozen-for-authorized-pass-b-replacement-and-later-adjudication',createdAt:new Date().toISOString(),blindPassPlan:fileRecord(`${prior}/execution-plan.json`),additionalWritablePaths:['pass-b','adjudication'].map(r=>`${next}/${r}-reading-receipt.json`),roles:['pass-b','adjudication'],instructions:[...guide.instructions,'All contexts and rationales must describe the actual source-specific argument; never generate placeholder text from move IDs, attach catalog-level templates to distinct moves, or treat an evidential weakness as automatically instantiating a named defect. Before first write, validate the complete draft in memory and finish normal drafting without stopping for routine approval.']});
write(`${next}/execution-plan.json`,{...oldPlan,status:'frozen-before-one-authorized-pass-b-replacement',authorization:fileRecord(`${next}/authorization.json`),freshContexts:1,totalBlindReviewContextsIncludingPreservedFailure:3,acceptedBlindReviewContexts:2,allowedInputs:[`${next}/packet.json`,`${next}/reference-definitions.json`,oldPacket.sourceTranscript.path,oldPacket.verifiedAudio.path].map(fileRecord),passes:[{pass:'pass-a',agentTask:oldPlan.passes[0].agentTask,outputPath:`${next}/pass-a-output.json`,attempts:1,retries:0,retainedWithoutNewAssessment:true,originalInput:fileRecord(`${prior}/packet.json`),originalOutput:fileRecord(`${prior}/pass-a-output.json`),originalReadingGate:fileRecord(`${prior}/pass-a-reading-gate.json`)},{pass:'pass-b',agentTask:`/root/rhetorical_${number}_b_replacement`,outputPath:`${next}/pass-b-output.json`,attempts:1,retries:0,originalFailedAttempts:1,authorizedReplacementAttempts:1}],controllerOverrides:{},sourcePacketBytesUnchanged:true,retainedFirstReviewBytesUnchanged:true});
write(`${next}/reading-continuation-review.json`,{status:'fresh-replacement-with-preserved-original-reading-incidents',debateNumber:number,originalReadingContinuation:fileRecord(`${prior}/reading-continuation-review.json`),originalPassBReceiptPathIncident:fileRecord(`${prior}/pass-b-receipt-path-incident.json`),retainedPassAReadingGate:fileRecord(`${prior}/pass-a-reading-gate.json`),retainedPassAInputsAreByteIdenticalAliases:true,originalPassBExcluded:true,replacementReadingPending:true,privateReasoningReadOrPreserved:false});
console.log(JSON.stringify({root:next,model,sourceViews:guide.sourceEvidenceViews.length,acceptedReviewContexts:2,newContextsAuthorized:1,directIncrementalCostUsd:0}));
