import fs from 'node:fs';
import assert from 'node:assert/strict';
import {pipeline} from 'node:stream/promises';
import {createHash} from 'node:crypto';
const plan=JSON.parse(fs.readFileSync('.assessment-cache/qhg-scope-audio-plan.json'));
const id=plan.videoId, cache=`.assessment-cache/captions/${id}`, out=`${cache}/original-audio.m4a`;
const caption=JSON.parse(fs.readFileSync(`${cache}/manifest.json`));
assert.equal(caption.videoId,id); assert(!fs.existsSync(out));
const html=await(await fetch(plan.canonicalUrl,{signal:AbortSignal.timeout(30000)})).text();
const key=html.match(/"INNERTUBE_API_KEY":"([^"]+)"/)?.[1];assert(key);
const p=await(await fetch(`https://www.youtube.com/youtubei/v1/player?key=${key}`,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({context:{client:{clientName:'ANDROID',clientVersion:'20.10.38',androidSdkVersion:30,hl:'en',gl:'US'}},videoId:id}),signal:AbortSignal.timeout(30000)})).json();
assert.equal(p.videoDetails.videoId,id);assert.equal(p.videoDetails.title,caption.title);
const candidates=p.streamingData.adaptiveFormats.filter(f=>f.itag===140&&f.url&&!f.audioTrack?.isAutoDubbed);
assert.equal(candidates.length,1);const f=candidates[0];
assert(!f.audioTrack||f.audioTrack.audioIsDefault);
const response=await fetch(f.url,{signal:AbortSignal.timeout(120000)});assert(response.ok,`Audio fetch HTTP ${response.status}`);
await pipeline(response.body,fs.createWriteStream(out,{flags:'wx'}));
const bytes=fs.readFileSync(out);
const record={videoId:id,canonicalUrl:plan.canonicalUrl,retrievedAt:new Date().toISOString(),acquisition:'Public YouTube Android player; single default audio stream, no alternative language track',format:{itag:f.itag,mimeType:f.mimeType,audioTrack:f.audioTrack??null,approxDurationMs:f.approxDurationMs},file:{path:out,bytes:bytes.length,sha256:createHash('sha256').update(bytes).digest('hex')},directCostUsd:0};
fs.writeFileSync(`${out}.provenance.json`,JSON.stringify(record,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify(record,null,2));
