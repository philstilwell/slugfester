import fs from 'node:fs';
import assert from 'node:assert/strict';
import {validateStandaloneCandidate} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
const wc=s=>String(s??'').trim().split(/\s+/).filter(Boolean).length;
export function checkPublication(o,p){
 const c=o.candidate,contract=p.outputRequirements;
 for(const k of ['schemaVersion','protocolId','status','debateNumber','debateId'])assert.equal(o[k],contract[k],k);
 const read=f=>JSON.parse(fs.readFileSync(f)),ledger=read(p.ledgerPath),scores=read(p.scorePath);
 validateStandaloneCandidate(c,scores);
 function compare(actual,frozen,key='candidate'){
  if(frozen===''){assert(typeof actual==='string'&&actual.trim(),key+' needs prose');return;}
  if(Array.isArray(frozen)){assert(Array.isArray(actual)&&actual.length===frozen.length,key+' array shape');frozen.forEach((v,i)=>compare(actual[i],v,key+'['+i+']'));return;}
  if(frozen&&typeof frozen==='object'){assert.deepEqual(Object.keys(actual).sort(),Object.keys(frozen).sort(),key+' keys');for(const k of Object.keys(frozen))compare(actual[k],frozen[k],key+'.'+k);return;}
  assert.deepEqual(actual,frozen,key+' frozen');
 }
 compare(c,p.lockedCandidateSkeleton);
 if(p.requiredReaderDisclosure)assert(c.sourceNote.includes(p.requiredReaderDisclosure),'Missing exact approved scope disclosure');
 const bounds=(s,low,high,label)=>{assert(wc(s)>=low&&(high==null||wc(s)<=high),label+' word count '+wc(s));assert(!/[\u3400-\u9fff\uac00-\ud7af\ufffd]/u.test(s),label+' script artifact');};
 bounds(c.summary,8,35,'summary');bounds(c.sourceNote,20,null,'source note');bounds(c.scoringNote,18,null,'scoring note');assert(/AI-generated/.test(c.scoringNote));
 const cards=c.sections.flatMap(s=>s.exchanges.flatMap(e=>['pro','con'].flatMap(side=>e[side]?[e[side]]:[])));
 for(const card of cards){bounds(card.role,1,5,card.ledgerMoveId+' role');bounds(card.words,8,55,card.ledgerMoveId+' description');bounds(card.critique,105,130,card.ledgerMoveId+' critique');assert(card.critique.length>=880,card.ledgerMoveId+' critique chars '+card.critique.length);const labels=['Strongest feature:','Principal limitation:','Live burden:','Locked score:'],parts=card.critique.split(/(?=Principal limitation:|Live burden:|Locked score:)/).map(s=>s.trim());assert(parts.length===4&&parts.every((v,i)=>v.startsWith(labels[i])&&/[.!?]$/.test(v)),card.ledgerMoveId+' labels');assert.equal((card.critique.match(/[.!?](?=\s|$)/g)||[]).length,4,card.ledgerMoveId+' sentences');assert.equal(card.tags.length,0);}
 assert(cards.reduce((s,c)=>s+wc(c.words),0)/cards.length>=p.argumentMinimumMean);assert(cards.filter(c=>wc(c.words)<20).length/cards.length<=.25);
 const tokens=cards.map(c=>c.critique.toLowerCase().replace(/[^a-z0-9/]+/g,' ').trim().split(/\s+/)),counts=new Map();
 for(const ts of tokens){const seen=new Set();for(let i=0;i<=ts.length-6;i++)seen.add(ts.slice(i,i+6).join(' '));for(const n of seen)counts.set(n,(counts.get(n)||0)+1);}
 const common=new Set([...counts].filter(([,n])=>n>=Math.max(3,Math.ceil(cards.length*.25))).map(([k])=>k));
 const ratios=tokens.map(ts=>{const covered=new Set();for(let i=0;i<=ts.length-6;i++)if(common.has(ts.slice(i,i+6).join(' ')))for(let j=i;j<i+6;j++)covered.add(j);return covered.size/ts.length;});
 assert(ratios.reduce((a,b)=>a+b,0)/ratios.length<=.15&&Math.max(...ratios)<=.25,'critique boilerplate');
 for(const side of ['pro','con']){bounds(c.quotes[side].text,3,18,side+' quote');bounds(c.quotes[side].context,12,55,side+' quote context');for(const s of c.overall[side].strengths)bounds(s,8,null,side+' strength');for(const b of c.overall[side].blunders){bounds(b.text,8,null,side+' blunder');assert.equal(b.links.length,0);}const a=c.logicalExtension[side];bounds(a.finalArgument.thesis,12,null,side+' thesis');a.finalArgument.premises.forEach(v=>bounds(v,12,null,side+' premise'));bounds(a.finalArgument.conclusion,15,null,side+' conclusion');for(const n of a.newArguments){bounds(n.title,2,8,side+' title');bounds(n.text,45,130,side+' reinforcing argument');}assert(!/AI-generated extension|AI-developed extension|possible extension|AI reconstruction|unassailable/i.test(JSON.stringify(a)),'AI provenance formula');}
 assert.deepEqual(o.sourceQuoteAudit.map(q=>q.side),['pro','con']);
 for(const q of o.sourceQuoteAudit){const m=ledger.moves.find(m=>m.moveId===q.moveId);assert(m&&m.side===q.side);assert.equal(q.speaker,c.sides[q.side].speaker);assert.equal(q.text,c.quotes[q.side].text);assert.equal(q.sourcePath,p.ledgerPath);assert.equal(q.transcriptPath,p.transcriptPath);assert(m.quoteEligibleExactSpans.includes(q.quoteEligibleExactSpan)&&q.quoteEligibleExactSpan.includes(q.text));for(const k of ['startEvent','endEvent','startMs','endMs'])assert.equal(q[k],m.sourceSpan[k]);assert.equal(q.exactMatch,true);}
 const paths=['pro','con'].flatMap(side=>['finalArgument',...c.logicalExtension[side].newArguments.map((_,i)=>'newArguments['+i+']')].map(field=>({side,field,path:'candidate.logicalExtension.'+side+'.'+field})));
 assert.deepEqual(o.noveltyMap.map(({side,field,path})=>({side,field,path})),paths);
 for(const n of o.noveltyMap){assert(n.hypothetical===true&&typeof n.explanation==='string'&&wc(n.explanation)>=12);assert(Array.isArray(n.sourceMoveIds)&&n.sourceMoveIds.every(id=>ledger.moves.some(m=>m.moveId===id)));}
 assert.deepEqual(o.audit,contract.audit);
 return{status:'mechanically-valid-editorial-review-required',cards:cards.length,sections:c.sections.length,minimumCritiqueCharacters:Math.min(...cards.map(x=>x.critique.length)),maximumCritiqueWords:Math.max(...cards.map(x=>wc(x.critique))),boilerplateMean:ratios.reduce((a,b)=>a+b,0)/ratios.length,boilerplateMax:Math.max(...ratios)};
}
if(process.argv[2]==='--check-stdin')console.log(JSON.stringify(checkPublication(JSON.parse(fs.readFileSync(0,'utf8')),JSON.parse(fs.readFileSync(process.argv[3])))));
