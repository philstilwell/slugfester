import fs from 'node:fs';import assert from 'node:assert/strict';
import {fileRecord} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
const n=process.argv[2],read=p=>JSON.parse(fs.readFileSync(p));assert.match(n,/^\d{3,}$/);
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
const original=r.root+'/judgments/pass-b',replacement=original+'/replacement-1';
assert.equal(fs.readFileSync(original+'/output.json','utf8'),'PLACEHOLDER\n');assert.equal(fs.readFileSync(original+'/reading-audit.json','utf8'),'PLACEHOLDER\n');
assert.equal(fs.existsSync(r.root+'/score-pass/output.json'),false);fs.mkdirSync(replacement,{recursive:false});
for(const name of ['output.json','reading-audit.json'])fs.copyFileSync(original+'/'+name,replacement+'/preserved-failed-'+name,fs.constants.COPYFILE_EXCL);
const instructions=read(original+'/execution-instructions.json');
instructions.outputPath=replacement+'/output.json';instructions.readingAuditPath=replacement+'/reading-audit.json';
instructions.requirements.push('This is the sole exceptional replacement after a prior context wrote no judgment content. Do not read the prior context, placeholder, failure record, other pass or other assessments. Author the first complete genuine judgment in the declared separate paths; never create a placeholder. No further replacement or retry is authorized.');
const put=(p,v)=>fs.writeFileSync(p,JSON.stringify(v,null,2)+'\n',{flag:'wx'});
put(replacement+'/execution-instructions.json',instructions);
put(replacement+'/authorization.json',{
 schemaVersion:'1.0-user-delegated-bounded-primary-replacement',status:'authorized-one-fresh-replacement-before-execution',debateNumber:n,debateId:r.debateId,createdAt:new Date().toISOString(),
 userAuthorization:'I am going to bed, so just use your discretion if you encounter anything that might trigger a request for approval.',
 controllerDecision:'Use one exceptional fresh isolated pass-b replacement for a wholly unavailable primary assessment. The failed context wrote only the literal PLACEHOLDER followed by a newline, with no dimension ratings, rationales or burden judgments. This does not reopen any accepted assessment substance. Preserve all failed bytes and the valid independent pass-a; if this replacement fails, stop rather than recursively replace it.',
 scope:'Only the same thirty-nine frozen moves and two burden-completion judgments, under identical allowed evidence, model, reasoning effort and rubric. No inventory, source, scope, cost cap, score control or other-pass changes.',
 originalInstructions:fileRecord(original+'/execution-instructions.json'),originalReadingRelease:fileRecord(original+'/reading-release.json'),failedOutput:fileRecord(replacement+'/preserved-failed-output.json'),failedReadingAudit:fileRecord(replacement+'/preserved-failed-reading-audit.json'),
 failure:{originalOutputPath:original+'/output.json',originalReadingAuditPath:original+'/reading-audit.json',actualBytesPerFile:12,substantiveJudgments:0,reason:'The original agent reported an accidental placeholder apply_patch and stopped without overwriting or retrying.'},
 replacementInstructions:fileRecord(replacement+'/execution-instructions.json'),replacementAttempts:1,retries:0,freshContextRequired:true,readingAuthenticationBeforeSubmissionRequired:true,canonicalInstallation:'After full schema and session authentication, mechanically replace canonical pass-b output and reading-audit with the separate successful replacement bytes; retained original instruction/release records are not overwritten.',directIncrementalCostUsd:0
});
console.log(JSON.stringify({status:'one-fresh-replacement-ready',instructions:replacement+'/execution-instructions.json',failedBytesPreserved:true}));
