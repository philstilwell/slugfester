import fs from 'node:fs';

import {execFileSync} from 'node:child_process';

console.log(JSON.stringify({status:report.status,earlierDebatesUnchanged:report.unchangedEarlierDebateObjects,archivedControllers:archived.length,removedControllers:report.removedControllerCount,removedDuplicateScreenshots:removedScreenshots.length,retainedRecorder}));
