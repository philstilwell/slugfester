import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {checkRepair} from './check-publication-field-repair.mjs';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1],group=args[args.indexOf('--group')+1];assert(args.includes('--debate')&&args.includes('--group'));assert.match(n,/^\d{3,}$/);assert.match(group,/^(cards-[1-9][0-9]*|overview|ai-contribution)$/);
const read=p=>JSON.parse(fs.readFileSync(p)),rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:createHash('sha256').update(b).digest('hex'),bytes:b.length};},r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
const dir=r.root+'/publication/field-repair-1/'+group,p=read(dir+'/execution-plan.json'),release=read(dir+'/reading-release.json');assert.equal(release.outputExistedAtRelease,false);assert(release.inputs.every(x=>x.literalCompleteReadAuthenticated));for(const x of p.allowedInputs)assert.equal(rec(x.path).sha256,x.sha256);
const session=rec(release.session.path),rows=fs.readFileSync(session.path,'utf8').trim().split('\n').map(JSON.parse),contexts=rows.filter(r=>r.type==='turn_context').map(r=>({model:r.payload.model,reasoningEffort:r.payload.effort}));assert(contexts.length&&contexts.every(c=>c.model===p.model.slug&&c.reasoningEffort===p.model.reasoningEffort));
const validation=checkRepair(read(p.outputPath),read(p.packetPath));const audit={schemaVersion:'1.0-authenticated-bounded-publication-field-repair',status:'passed-authenticated-first-field-repair',debateNumber:n,debateId:r.debateId,groupId:group,model:p.model,actualModelContexts:contexts,agentPath:release.agentPath,session:{...session,kind:'prefix-snapshot-after-output'},executionPlan:rec(dir+'/execution-plan.json'),readingRelease:rec(dir+'/reading-release.json'),readingAudit:rec(p.readingAuditPath),output:rec(p.outputPath),shards:p.shards,validation,attemptsPerShard:1,retries:0,protocolIsolation:true,operatingSystemFilesystemIsolation:false,scoreChanges:0,judgmentChanges:0,directIncrementalCostUsd:0};fs.writeFileSync(dir+'/execution.json',JSON.stringify(audit,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({status:audit.status,groupId:group,validation}));
