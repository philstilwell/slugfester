import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {checkRepair} from './check-publication-field-repair.mjs';import {checkPublication} from './check-publication-draft.mjs';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1],mode=args[args.indexOf('--mode')+1];assert(args.includes('--debate')&&args.includes('--mode')&&/^\d{3,}$/.test(n));assert(['authenticate','assemble','register'].includes(mode));const read=p=>JSON.parse(fs.readFileSync(p)),rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:createHash('sha256').update(b).digest('hex'),bytes:b.length};},put=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n',{flag:'wx'});
const rp='docs/assessment-production/standalone-debates-v1/registry.json',registry=read(rp),r=registry.debates.find(x=>x.debateNumber===n);assert(r);const pub=r.root+'/publication',dir=pub+'/exceptional-editorial-repair-1',p=read(dir+'/execution-plan.json'),packet=read(p.packetPath),diag=read(pub+'/field-repair-1/failure-preservation/diagnostic.json');
if(mode==='authenticate'){
 const release=read(dir+'/reading-release.json');assert.equal(release.outputExistedAtRelease,false);assert(release.inputs.every(x=>x.literalCompleteReadAuthenticated));for(const x of p.allowedInputs)assert.equal(rec(x.path).sha256,x.sha256);const rows=fs.readFileSync(release.session.path,'utf8').trim().split('\n').map(JSON.parse),contexts=rows.filter(x=>x.type==='turn_context').map(x=>({model:x.payload.model,reasoningEffort:x.payload.effort}));assert(contexts.length&&contexts.every(x=>x.model===p.model.slug&&x.reasoningEffort===p.model.reasoningEffort));
 const writes=rows.flatMap((row,i)=>{const x=row.payload;if(!['function_call','custom_tool_call'].includes(x?.type))return[];const text=x.input??x.arguments??'';if(!text.includes(p.outputPath)||!text.includes('*** '))return[];return[{rowIndex:i,callId:x.call_id,add:text.includes('*** Add File: '+process.cwd()+'/'+p.outputPath),update:text.includes('*** Update File: '+process.cwd()+'/'+p.outputPath)}];});assert.equal(writes.filter(w=>w.add).length,1);assert.equal(writes.filter(w=>w.update).length,0,'No post-submission overwrite permitted');
 const validation=checkRepair(read(p.outputPath),packet);put(dir+'/execution.json',{status:'passed-authenticated-user-authorized-exceptional-editorial-repair',debateNumber:n,debateId:r.debateId,model:p.model,agentPath:release.agentPath,actualModelContexts:contexts,session:{...rec(release.session.path),kind:'prefix-snapshot-after-first-submission'},readingRelease:rec(dir+'/reading-release.json'),readingAudit:rec(p.readingAuditPath),plan:rec(dir+'/execution-plan.json'),authorization:rec(dir+'/authorization.json'),output:rec(p.outputPath),shards:p.shards,submissionWrites:writes,validation,attemptsPerShard:1,retries:0,priorUnauthorizedRevisionsUsed:false,judgmentChanges:0,scoreChanges:0,directIncrementalCostUsd:0});console.log(JSON.stringify({status:'exceptional-submission-authenticated',validation}));
}else if(mode==='assemble'){
 const e=read(dir+'/execution.json'),accepted=read(p.outputPath),notes=read(dir+'/controller-editorial-acceptance.json');assert.equal(e.output.sha256,rec(p.outputPath).sha256);assert.equal(notes.status,'passed-direct-source-specific-repair-review');assert.equal(notes.outputSha256,e.output.sha256);checkRepair(accepted,packet);assert.equal(rec(pub+'/output.json').sha256,diag.publicationUnchanged.sha256);
 const replacements=accepted.shards.flatMap(s=>s.fields);assert.deepEqual(replacements.map(f=>f.path),diag.rejectedFields.map(f=>f.path));const retained=diag.retainedValidFields.map(f=>({path:f.path,text:f.submittedText})),all=[...retained,...replacements],o=read(pub+'/output.json');assert.equal(new Set(all.map(f=>f.path)).size,all.length);
 for(const f of all){const keys=f.path.replace(/\[(\d+)\]/g,'.$1').split('.');let target=o;for(const k of keys.slice(0,-1))target=target[k];assert.notEqual(target[keys.at(-1)],f.text);target[keys.at(-1)]=f.text;}
 for(const g of diag.groups){const gp=read(g.executionPlan.path),go=read(g.submittedOutput.path);for(const s of go.shards)for(const f of s.fields){const replacement=replacements.find(x=>x.path===f.path);if(replacement)f.text=replacement.text;}checkRepair(go,read(gp.packetPath));}
 const validation=checkPublication(o,read(pub+'/input-bundle.json'));put(dir+'/accepted-publication.json',o);
 const shards=[];for(const g of diag.groups){const gp=read(g.executionPlan.path);for(const s of gp.shards){const fields=s.fields.filter(f=>retained.some(x=>x.path===f));if(fields.length)shards.push({shardId:g.groupId+'-'+s.shardId+'-retained',writableFieldCount:fields.length,fields,input:rec(gp.packetPath),output:g.submittedOutput,attempts:1,retries:0,disposition:'retained-first-submission-valid-leaves'});}}for(const s of p.shards)shards.push({shardId:'exceptional-'+s.shardId,writableFieldCount:s.fields.length,fields:s.fields,input:rec(p.packetPath),output:rec(p.outputPath),attempts:1,retries:0,disposition:'user-authorized-one-additional-attempt',authorization:rec(dir+'/authorization.json')});
 put(dir+'/combined-execution.json',{status:'passed-isolated-card-depth-repair',debateNumber:n,debateId:r.debateId,model:{label:p.model.displayLabel,slug:p.model.slug,reasoningEffort:p.model.reasoningEffort},firstAttemptDiagnostic:rec(pub+'/field-repair-1/failure-preservation/diagnostic.json'),exceptionalExecution:rec(dir+'/execution.json'),shards,audit:{judgmentChanges:0,scoreChanges:0,moveChanges:0,tagChanges:0,retainedValidFields:retained.length,exceptionallyRepairedFields:replacements.length,maximumWritableFieldsPerShard:2},directIncrementalCostUsd:0});
 put(dir+'/assembly-audit.json',{status:'passed-user-authorized-bounded-assembly',debateNumber:n,debateId:r.debateId,before:diag.publicationUnchanged,after:rec(dir+'/accepted-publication.json'),retainedFieldPaths:retained.map(f=>f.path),replacementFieldPaths:replacements.map(f=>f.path),onlyDeclaredLeavesChanged:true,priorUnauthorizedRevisionsUsed:false,validation,judgmentChanges:0,scoreChanges:0,moveChanges:0});fs.copyFileSync(dir+'/accepted-publication.json',pub+'/output.json');console.log(JSON.stringify({status:'accepted-publication-assembled',validation}));
}else{
 const before=rec(pub+'/field-repair-1/preserved-prior-output.json'),after=rec(dir+'/accepted-publication.json'),path=pub+'/correction-content-parity.json';put(path,{schemaVersion:'1.0-standalone-content-parity-correction',status:'applied-once-and-frozen',debateNumber:n,debateId:r.debateId,preservedPriorOutput:before,evidence:{before,after,contentParityAudit:rec(pub+'/content-parity-audit.json'),execution:rec(dir+'/combined-execution.json')},writableShards:read(dir+'/combined-execution.json').shards,audit:{judgmentChanges:0,scoreChanges:0,moveChanges:0,tagChanges:0,maximumWritableFieldsPerShard:2},directIncrementalCostUsd:0});r.contentParity={auditPath:pub+'/content-parity-audit.json',correctionPath:path,preservedOutputPath:before.path,executionPath:dir+'/combined-execution.json'};fs.writeFileSync(rp,JSON.stringify(registry,null,2)+'\n');console.log('Bounded editorial correction chain registered.');
}
