import fs from 'node:fs';
import vm from 'node:vm';
import assert from 'node:assert/strict';
import {fileRecord} from '../scripts/lib/assessment-production-standalone-debate-v1.mjs';
import {debateSummaries} from '../src/data/debate-summaries.js';
const args=process.argv.slice(2),n=args[args.indexOf('--debate')+1],port=Number(args[args.indexOf('--port')+1]);assert(args.includes('--debate')&&args.includes('--port'));assert.match(n,/^\d{3,}$/);assert(Number.isInteger(port)&&port>1024&&port<65536);
const read=p=>JSON.parse(fs.readFileSync(p)),r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);assert(debateSummaries.some(d=>d.number===n));
const testPath='tests/site-quality/site-quality.spec.mjs',text=fs.readFileSync(testPath,'utf8'),budgetText=text.match(/const catalogueBudget = (\{[\s\S]*?\n\});/)[1],routesText=text.match(/const routeBudgets = (\[[\s\S]*?\n\]);/)[1];
const budget=vm.runInNewContext('('+budgetText+')',{}, {timeout:1000}),routes=vm.runInNewContext('('+routesText+')',{}, {timeout:1000}),added=Math.max(0,debateSummaries.length-budget.baselineDebates);assert(added<=budget.reviewAfterAddedDebates);
const expected={debateNumber:n,debateId:r.debateId,origin:'http://127.0.0.1:'+port,publishedDebates:debateSummaries.length,budget,addedDebates:added,test:fileRecord(testPath),routes:routes.map(x=>({...x,limitBytes:x.baseDataBytes+added*budget.bytesPerAddedDebate}))};
const body=async(page)=>{const browser=page.context().browser();if(!browser)throw Error('Browser unavailable');const measurements=[];for(const target of expected.routes){const context=await browser.newContext({viewport:{width:1440,height:1000},serviceWorkers:'block'});try{const p=await context.newPage();await p.goto(expected.origin+target.route,{waitUntil:'networkidle'});await p.locator('main h1').first().waitFor();const resources=await p.evaluate(()=>performance.getEntriesByType('resource').filter(e=>e.name.includes('/src/data/')).map(e=>({name:new URL(e.name).pathname,bytes:e.decodedBodySize})));const loadedBytes=resources.reduce((s,x)=>s+x.bytes,0);if(!resources.some(x=>x.name.endsWith(target.required))||resources.some(x=>x.name.endsWith('/src/data/debates.js'))||loadedBytes>target.limitBytes)throw Error(JSON.stringify({target,loadedBytes,resources}));measurements.push({route:target.route,resources,loadedBytes,limitBytes:target.limitBytes,headroomBytes:target.limitBytes-loadedBytes});}finally{await context.close();}}return {status:'passed-measured-review-without-budget-changes',...expected,measurementMethod:'fresh-isolated-browser-context-per-route',measurements,baselineChanged:false,limitsChanged:false,temporaryMeasurementContextsRemaining:0};};
const out='output/playwright/debate-'+n;fs.mkdirSync(out,{recursive:true});fs.writeFileSync(out+'/performance-review.js','async (page) => { const expected='+JSON.stringify(expected)+'; return ('+body.toString()+')(page); }\n',{flag:'wx'});
fs.mkdirSync(r.root+'/validation',{recursive:true});fs.writeFileSync(r.root+'/validation/performance-review-plan.json',JSON.stringify({...expected,status:'frozen-read-only-measurement-plan',baselineMutationAuthorized:false,directIncrementalCostUsd:0},null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({status:'prepared-unchanged-budget-measurement',expected}));
