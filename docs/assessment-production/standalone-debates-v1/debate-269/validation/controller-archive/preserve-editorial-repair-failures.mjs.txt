import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {checkRepair} from './check-publication-field-repair.mjs';
const a=process.argv.slice(2),n=a[a.indexOf('--debate')+1];assert(a.includes('--debate')&&/^\d{3,}$/.test(n));const read=p=>JSON.parse(fs.readFileSync(p)),rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:createHash('sha256').update(b).digest('hex'),bytes:b.length};},put=(p,o)=>fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n',{flag:'wx'});
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);const base=r.root+'/publication/field-repair-1',dir=base+'/failure-preservation';fs.mkdirSync(dir,{recursive:true});const groups=[],invalid=[],valid=[],wc=s=>s.trim().split(/\s+/).length;
for(const groupId of ['cards-1','cards-2','cards-3']){const g=base+'/'+groupId,p=read(g+'/packet.json'),submitted=groupId==='cards-3'?dir+'/cards-3-first-submitted-output.json':g+'/output.json',o=read(submitted);const dest=dir+'/'+groupId+'-submitted-output.json';if(!fs.existsSync(dest))fs.copyFileSync(g+'/output.json',dest,fs.constants.COPYFILE_EXCL);else assert.equal(rec(dest).sha256,rec(g+'/output.json').sha256);let validation;try{validation=checkRepair(o,p);}catch(e){validation={status:'failed',message:e.message,stack:e.stack};}const plan=read(g+'/execution-plan.json'),release=read(g+'/reading-release.json');assert.equal(release.outputExistedAtRelease,false);assert(release.inputs.every(x=>x.literalCompleteReadAuthenticated));for(const x of plan.allowedInputs)assert.equal(rec(x.path).sha256,x.sha256);const session=rec(release.session.path),rows=fs.readFileSync(session.path,'utf8').trim().split('\n').map(JSON.parse),contexts=rows.filter(x=>x.type==='turn_context').map(x=>({model:x.payload.model,reasoningEffort:x.payload.effort}));assert(contexts.every(x=>x.model===plan.model.slug&&x.reasoningEffort===plan.model.reasoningEffort));groups.push({groupId,validation,submittedOutput:rec(submitted),observedCurrentOutput:rec(g+'/output.json'),preservedObservedOutput:rec(dest),postSubmissionMutationQuarantined:groupId==='cards-3',readingAudit:rec(g+'/reading-audit.json'),executionPlan:rec(g+'/execution-plan.json'),readingRelease:rec(g+'/reading-release.json'),session:{...session,kind:'prefix-snapshot-after-submission'},actualModelContexts:contexts,attemptsPerShard:1,authorizedRetries:0});
for(const s of o.shards)for(const f of s.fields){const requirement=p.shards.find(x=>x.shardId===s.shardId).fields.find(x=>x.path===f.path),failures=[],words=wc(f.text),characters=f.text.length;if(words<requirement.minimumWords||words>requirement.maximumWords)failures.push(`Word count ${words} outside frozen ${requirement.minimumWords}–${requirement.maximumWords} repair range.`);if(characters<(requirement.minimumCharacters??0))failures.push('Below minimum character depth.');if(requirement.kind==='critique'){const parts=f.text.split(/(?=Principal limitation:|Live burden:|Locked score:)/).map(s=>s.trim()),labels=['Strongest feature:','Principal limitation:','Live burden:','Locked score:'];if(parts.length!==4||!parts.every((s,i)=>s.startsWith(labels[i])&&/[.!?]$/.test(s))||(f.text.match(/[.!?](?=\s|$)/g)||[]).length!==4)failures.push('Four labeled sentence contract failed.');}
if(groupId==='cards-2'&&s.shardId==='014')failures.push('The phrase caused beginning narrows the frozen disjunction a cause or a beginning; preserve Linford’s nonacceptance burden without requiring that competing alternatives be disproved or established.');
const row={groupId,shardId:s.shardId,path:f.path,moveId:p.shards.find(x=>x.shardId===s.shardId).moveId??null,words,characters,requiredMinimumWords:requirement.minimumWords,requiredMaximumWords:requirement.maximumWords,requiredMinimumCharacters:requirement.minimumCharacters??null,failures,submittedText:f.text};(failures.length?invalid:valid).push(row);}}
assert.equal(invalid.length,14);assert.equal(valid.length,57);
put(dir+'/diagnostic.json',{status:'paused-after-exhausted-first-editorial-repair-fields',debateNumber:n,debateId:r.debateId,groups,rejectedFields:invalid,retainedValidFields:valid,counts:{submittedFields:invalid.length+valid.length,validFields:valid.length,rejectedFields:invalid.length,overlengthCritiques:invalid.filter(x=>x.path.endsWith('.critique')).length},publicationUnchanged:rec(r.root+'/publication/output.json'),scores:read(r.root+'/score-pass/output.json').overall,subsequentRhetoricalReviewerCalls:0,siteIntegrated:false,committed:false,pushed:false,directIncrementalCostUsd:0});
put(dir+'/proposed-exception.json',{status:'awaiting-user-authorization-not-executed',debateNumber:n,debateId:r.debateId,requestedAction:'One additional bounded wording-repair attempt for only the fourteen rejected leaves, retaining all fifty-seven valid replacement leaves and every frozen assessment artifact.',fields:invalid.map(({groupId,shardId,path,moveId,failures})=>({groupId,originalShardId:shardId,path,moveId,failures})),maximumWritableLeafFieldsPerShard:2,additionalAttemptsPerRejectedField:1,mayChangeEvidence:false,mayChangeScores:false,mayChangeJudgments:false,mayAlterPassingFields:false,directIncrementalCostUsd:0,diagnostic:rec(dir+'/diagnostic.json')});
console.log(JSON.stringify({status:'preserved-and-paused',validFields:valid.length,rejectedFields:invalid.length,diagnostic:dir+'/diagnostic.json'}));
