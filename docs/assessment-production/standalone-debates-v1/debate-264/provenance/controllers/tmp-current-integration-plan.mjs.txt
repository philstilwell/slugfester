import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import crypto from 'node:crypto';
import { debates } from './src/data/debates.js';
const read = p => JSON.parse(fs.readFileSync(p, 'utf8'));
const sha = p => crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const serialize = o => JSON.stringify(o, null, 2) + '\n';
export function integrationPlan(number) {
  const registry = read('docs/assessment-production/standalone-debates-v1/registry.json');
  const rec = registry.debates.find(r => r.debateNumber === number);
  assert(rec && rec.status !== 'published-and-frozen');
  const publicationPath = rec.root + '/publication/output.json';
  const c = read(publicationPath).candidate;
  assert.equal(c.number, number); assert.equal(c.id, rec.debateId);
  assert.equal(new URL(c.youtubeUrl).searchParams.get('v'), rec.videoId);
  assert(!debates.some(d => d.number === number || d.id === c.id || new URL(d.youtubeUrl).searchParams.get('v') === rec.videoId));
  const editorial = read(rec.root + '/publication/controller-review.json');
  assert.equal(editorial.status, 'accepted-for-rhetorical-review');
  const ret = rec.root + '/publication/rhetorical-tag-review-1';
  const tags = read(ret + '/audit.json');
  assert.equal(tags.status, 'passed-rhetorical-tag-review');
  assert.equal(tags.debateId, rec.debateId);
  const source = read('.assessment-cache/captions/' + rec.videoId + '/manifest.json');
  const transcriptPath = '.assessment-cache/captions/' + rec.videoId + '/transcript.txt';
  const eventsPath = '.assessment-cache/captions/' + rec.videoId + '/events.json';
  assert.equal(source.transcriptSha256, sha(transcriptPath));
  assert.equal(source.normalizedEventsSha256, sha(eventsPath));
  const corpusPath = 'docs/calibration/v2.1/corpus-transcript-audit.json';
  const corpus = read(corpusPath);
  assert(!corpus.entries.some(e => e.debateId === c.id || e.videoId === rec.videoId));
  const wordCount = fs.readFileSync(transcriptPath, 'utf8').trim().split(/\s+/).length;
  const events = read(eventsPath);
  assert(Array.isArray(events)); assert.equal(events.length, source.eventCount);
  corpus.entries.push({debateNumber:number, debateId:c.id, videoId:rec.videoId, status:'available', transcriptStorage:transcriptPath, eventsStorage:eventsPath, track:source.track, extractionMethod:source.extractionMethod, audioSourceUrl:null, model:null, wordCount, spokenWordCount:events.map(e=>e.text).join(' ').trim().split(/\s+/).length, eventCount:source.eventCount, retrievedAt:source.retrievedAt, transcriptSha256:source.transcriptSha256, normalizedEventsSha256:source.normalizedEventsSha256});
  corpus.corpusDebates = corpus.entries.length;
  corpus.locallySavedTranscripts = corpus.entries.filter(e=>e.status === 'available').length;
  corpus.unavailableTranscripts = corpus.entries.filter(e=>e.status !== 'available').length;
  corpus.acquisitionRate = corpus.locallySavedTranscripts / corpus.corpusDebates;
  corpus.auditAsOf = new Date().toISOString();
  const debatePath = 'src/data/debates.js', priorDebates = fs.readFileSync(debatePath,'utf8');
  const marker = '\n];\n\naddMissingAiContributions(debates);';
  assert.equal(priorDebates.split(marker).length, 2);
  const inserted = ',\n' + JSON.stringify(c,null,2).split('\n').map(l=>'  '+l).join('\n');
  const nextDebates = priorDebates.replace(marker, inserted + marker);
  const bio = read(rec.root + '/provenance/biography-preparation.json');
  assert.equal(bio.status,'verified-background-only-not-assessment-evidence');
  assert([c.sides.pro.speaker,c.sides.con.speaker].includes(bio.speaker));
  const bioPath = 'src/data/interlocutor-bios.js', oldBio = fs.readFileSync(bioPath,'utf8');
  assert(!oldBio.includes('['+JSON.stringify(bio.speaker)+','));
  const newBio = oldBio.replace('const entries = [\n','const entries = [\n  '+JSON.stringify([bio.speaker,bio.proposedText,bio.source,bio.reviewed])+',\n');
  const changes = [{path:debatePath,before:priorDebates,after:nextDebates},{path:bioPath,before:oldBio,after:newBio},{path:corpusPath,before:fs.readFileSync(corpusPath,'utf8'),after:serialize(corpus)}];
  return {debateNumber:number,publicationSha256:sha(publicationPath),changes};
}
if (process.argv[2] === '--debate') console.log(JSON.stringify(integrationPlan(process.argv[3])));
