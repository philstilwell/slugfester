import fs from 'node:fs';

import crypto from 'node:crypto';
import assert from 'node:assert/strict';
import {validateStandalonePrimaryJudgment} from './scripts/lib/assessment-production-standalone-debate-v1.mjs';
const read=p=>JSON.parse(fs.readFileSync(p));
const record=p=>{const bytes=fs.readFileSync(p);return{path:p,sha256:crypto.createHash('sha256').update(bytes).digest('hex'),bytes:bytes.length};};
const write=(p,x)=>{assert(!fs.existsSync(p),'Preserve existing '+p);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,JSON.stringify(x,null,2)+'\n',{flag:'wx'});};
const [flag,n]=process.argv.slice(2);assert.equal(flag,'--debate');
const route=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(r=>r.debateNumber===n);assert(route);
const b=route.root,root=b+'/judgments/exceptional-replacement',plan=read(root+'/execution-plan.json'),inventory=read(plan.inventory.path),manifest=read(b+'/manifest.json');

assert.equal(precisionReview.status,'accepted-two-dimension-replacement');assert.equal(record(precisionTask.outputPath).sha256,precisionReview.outputSha256);
const built=[];
for(const pass of ['pass-a','pass-b']){
 const tasks=plan.tasks.filter(t=>t.pass===pass),parts=tasks.map(task=>{
  const dir=path.dirname(task.packet.path),auth=read(dir+'/output-authentication.json'),review=read(dir+'/controller-review.json');
  assert.equal(auth.status,'authenticated-output-awaiting-semantic-review');assert.equal(record(task.outputPath).sha256,auth.output.sha256);assert.equal(record(task.outputPath).sha256,review.outputSha256);
  const output=read(task.outputPath);let acceptedOutput=structuredClone(output),override=null;
  if(pass==='pass-b'&&task.sectionId===precisionOutput.sectionId){throw new Error('Unexpected overlapping sectionId in precision output; derive match only from exact move IDs');}
  const targetIds=precisionOutput.replacements.map(r=>r.moveId);
  const needsPrecision=pass==='pass-b'&&task.kind==='section'&&targetIds.every(id=>output.judgments.some(j=>j.moveId===id));
  if(needsPrecision){
   assert.equal(review.status,'rejected-two-precision-fields-caption-error-contamination');
   for(const item of precisionOutput.replacements){const j=acceptedOutput.judgments.find(x=>x.moveId===item.moveId);j.dimensions[item.dimension]={value:item.value,rationale:item.rationale};}
   const restored=structuredClone(acceptedOutput);for(const item of precisionOutput.replacements)restored.judgments.find(j=>j.moveId===item.moveId).dimensions[item.dimension]=output.judgments.find(j=>j.moveId===item.moveId).dimensions[item.dimension];assert.deepEqual(restored,output);
   override={authorization:record(precisionRoot+'/authorization.json'),packet:precisionTask.packet,output:record(precisionTask.outputPath),authentication:record(precisionRoot+'/'+precisionTask.assignmentId+'/output-authentication.json'),review:record(precisionRoot+'/'+precisionTask.assignmentId+'/controller-review.json'),fields:precisionOutput.replacements.map(r=>({moveId:r.moveId,dimension:r.dimension})),allOtherFieldsUnchanged:true};
  }else assert(['accepted-section-judgment','accepted-burden-judgment'].includes(review.status),task.assignmentId);
  return{task,output:acceptedOutput,evidence:{assignmentId:task.assignmentId,packet:task.packet,output:record(task.outputPath),authentication:record(dir+'/output-authentication.json'),review:record(dir+'/controller-review.json'),override}};
 });
 const sectionParts=parts.filter(p=>p.task.kind==='section'),burden=parts.find(p=>p.task.kind==='burden');assert.equal(sectionParts.length,inventory.sections.length);assert(burden);
 const js=sectionParts.flatMap(p=>p.output.judgments),checks=sectionParts.flatMap(p=>p.output.sourceReview.checks);assert.equal(js.length,inventory.moves.length);assert.equal(new Set(js.map(j=>j.moveId)).size,inventory.moves.length);

 const oldPath=b+'/judgments/'+pass+'/output.json',oldAuth=read(b+'/judgments/'+pass+'/controller-audit.json');assert.equal(record(oldPath).sha256,oldAuth.output.sha256);
 built.push({pass,judgment,validation,parts:parts.map(p=>p.evidence),oldPath,oldHash:oldAuth.output.sha256});
}
const passes=[];
for(const x of built){
 const assembled=root+'/'+x.pass+'/assembled-output.json';write(assembled,x.judgment);
 const preserved=b+'/judgments/'+x.pass+'/failed-initial-output.json';assert(!fs.existsSync(preserved));fs.copyFileSync(x.oldPath,preserved,fs.constants.COPYFILE_EXCL);assert.equal(record(preserved).sha256,x.oldHash);
 fs.copyFileSync(assembled,x.oldPath);assert.equal(record(assembled).sha256,record(x.oldPath).sha256);
