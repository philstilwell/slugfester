import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import crypto from 'node:crypto';
const read=p=>JSON.parse(fs.readFileSync(p));
const rec=p=>{const b=fs.readFileSync(p);return{path:p,sha256:crypto.createHash('sha256').update(b).digest('hex'),bytes:b.length};};
const write=(p,x)=>{assert(!fs.existsSync(p));fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,JSON.stringify(x,null,2)+'\n',{flag:'wx'});};
const [flag,n]=process.argv.slice(2);assert.equal(flag,'--debate');
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(x=>x.debateNumber===n);assert(r);
const b=r.root,root=b+'/judgments/precision-replacement',prior=b+'/judgments/exceptional-replacement',old=read(prior+'/execution-plan.json'),proposalPath=prior+'/two-field-recovery-proposal.json',proposal=read(proposalPath),m=read(b+'/manifest.json');
for(const lock of [...Object.values(m.sourceLocks),...Object.values(m.controlLocks),old.inventory,...old.readingFiles])assert.equal(rec(lock.path).sha256,lock.sha256);
const original=prior+'/pass-b--purpose-and-fine-tuning/output.json',review=read(prior+'/pass-b--purpose-and-fine-tuning/controller-review.json');assert.equal(rec(original).sha256,review.outputSha256);
const assignmentId='pass-b--two-clarity-fields',packetPath=root+'/'+assignmentId+'/packet.json',outputPath=path.resolve(root,assignmentId,'output.json');
const fields=proposal.proposedReplacement.writableDimensionRecords;
const sourceFacts=read(b+'/audio/retained-move-resolutions.json').findings.filter(f=>fields.some(x=>x.moveId===f.moveId));assert.equal(sourceFacts.length,2);
const packet=structuredClone(read(prior+'/pass-b--purpose-and-fine-tuning/packet.json'));
packet.assignmentId=assignmentId;packet.kind='dimension';packet.expectedMoveIds=fields.map(f=>f.moveId);packet.writableDimensionRecords=fields;packet.outputPath=outputPath;packet.factualAudioResolutions=sourceFacts;
packet.requirements.push('This exceptional task replaces exactly two precisionClarity dimension records. Judge stable terms, distinctions, scope, and recoverable reasoning in the speaker’s actual performance. Do not judge truth or evidential support under precision. Caption recognition errors are not participant defects and must not affect the value; the attached completed audio-derived factual resolutions resolve the listed terms. They do not establish the scientific truth of any claim. Return no other ratings, burden adjustments, sourceReview or edits. Do not inspect the prior output or its rejection record.');
const s=packet.outputShape;delete s.judgments;delete s.sourceReview;s.assignmentId=assignmentId;s.kind='dimension';s.replacements=fields.map(f=>({...f,value:'integer 0..100',rationale:'source-grounded precision/clarity rationale, at least 40 characters'}));
write(root+'/authorization.json',{schemaVersion:'1.0-authorized-two-dimension-replacement',status:'authorized-and-frozen-before-execution',debateNumber:n,debateId:r.debateId,authorizedAt:new Date().toISOString(),userResponse:'Yes.',approvedRequest:'One fresh isolated reviewer replaces only two affected pass-b clarity judgments, preserving all other fields, then continue the already-approved work; additional direct cost $0.',proposal:rec(proposalPath),preservedOriginal:rec(original),writableDimensionRecords:fields,attempts:1,retries:0,model:old.model,inventory:old.inventory,source:old.source,additionalPaidCallsPermitted:false});
write(packetPath,packet);
write(root+'/execution-plan.json',{...old,status:'frozen-before-execution',frozenAt:new Date().toISOString(),authorization:rec(root+'/authorization.json'),tasks:[{assignmentId,pass:'pass-b',kind:'dimension',sectionId:packet.sectionId,moveCount:fields.length,packet:rec(packetPath),outputPath,attempts:1,retries:0}],maximumConcurrency:1,sourceFactBasis:rec(b+'/audio/retained-move-resolutions.json')});
console.log(JSON.stringify({packetPath,outputPath,readingChunks:packet.readingFiles.length,fields,directCostUsd:0}));
