import fs from 'node:fs';
import assert from 'node:assert/strict';
import crypto from 'node:crypto';
const read=p=>JSON.parse(fs.readFileSync(p));
const record=p=>{const b=fs.readFileSync(p);return{path:p,sha256:crypto.createHash('sha256').update(b).digest('hex'),bytes:b.length};};
const [flag,n]=process.argv.slice(2);assert.equal(flag,'--debate');
const r=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(r=>r.debateNumber===n);assert(r);
const b=r.root,a=b+'/audio',invPath=b+'/inventory/inventory.json',inventory=read(invPath),execution=read(b+'/judgments/execution.json');assert.equal(execution.status,'complete-and-schema-valid');
const plan=read(a+'/plan.json'),facts=read(a+'/retained-move-resolutions.json'),cost=read(a+'/cost-reconciliation.json'),notes=read(a+'/controller-clip-notes.json');
assert.equal(plan.videoId,r.videoId);assert.equal(facts.debateId,r.debateId);assert.equal(cost.debateId,r.debateId);
assert.equal(notes.reviews.length,plan.clips.length);assert.equal(cost.failedOrTransportUncertainCalls,0);assert.equal(cost.retries,0);assert(cost.withinCap);
for(const call of cost.calls)for(const key of ['response','attempt','completion'])if(call[key])assert.equal(record(call[key].path).sha256,call[key].sha256);
assert.equal(record(plan.sourceMedia.path).sha256,plan.sourceMedia.sha256);
for(const clip of plan.clips)assert.equal(record(clip.path).sha256,clip.sha256);
const passes=execution.passes.map(p=>{assert.equal(record(p.outputPath).sha256,p.outputSha256);return read(p.outputPath);});
const triggeredMoveIds=inventory.moves.filter(m=>m.attributionConfidence!=='high').map(m=>m.moveId);
const primaryConcerns=passes.flatMap(p=>p.sourceReview.checks.filter(c=>!c.attributionSupported||c.materialUncertainty).map(c=>({pass:p.pass,...c})));
const findings=new Map(facts.findings.map(f=>[f.moveId,f]));
for(const id of [...triggeredMoveIds,...primaryConcerns.map(c=>c.moveId)]){const f=findings.get(id);assert(f&&f.materialMeaningResolved&&f.retainedQuotationOwnershipResolved,'Unresolved source concern '+id);}
const buildCheck=id=>{const m=inventory.moves.find(m=>m.moveId===id),clips=plan.clips.filter(c=>c.moveIds.includes(id));assert(m&&clips.length);return{moveId:id,sourceEventRange:[m.sourceSpan.startEvent,m.sourceSpan.endEvent],speaker:m.speaker,retainedAttributionConfidence:m.attributionConfidence,method:'Controller-reviewed independent audio-derived transcription with known-speaker references; visual turn corroboration where automatic labels conflicted. No direct listening claimed.',resolution:findings.get(id),verificationClips:clips.map(c=>({clipId:c.clipId,startMs:c.startMs,endMs:c.endMs,response:record(a+'/responses/'+c.clipId+'.json'),completion:record(a+'/responses/'+c.clipId+'.completion.json'),controllerNote:notes.reviews.find(x=>x.clipId===c.clipId).result})),sourceAndInventoryUnchanged:true};};
const checks=triggeredMoveIds.map(buildCheck),supplementalChecks=plan.selectedMoveIds.filter(id=>!triggeredMoveIds.includes(id)).map(buildCheck);
const out={schemaVersion:'1.0-standalone-audio-verification',protocolId:'assessment-production-standalone-debate-v1',status:'complete',debateNumber:n,debateId:r.debateId,videoId:r.videoId,inventory:record(invPath),triggeredMoveIds,checks,supplementalChecks,audioAcquired:true,audioListenedTo:false,paidTranscriptionPerformed:true,fullDebateAudioVerified:false,disposition:'Five below-high-attribution inventory triggers and supplementary retained-move wording concerns are resolved through the preserved 15 planned verification clips, two earlier source-context clips and three visual contact sheets. Automatic speaker labels were not imported blindly. The two wrong-speaker quotations were separately removed with explicit approval before inventory freeze. No excluded moderator material is scoring evidence; no source caption bytes or frozen inventory fields were changed by this gate.',independentReviews:execution.passes.map(p=>record(p.outputPath)),primaryConcernMapping:primaryConcerns.map(c=>({...c,resolution:findings.get(c.moveId)})),verificationPlan:record(a+'/plan.json'),factualResolutions:record(a+'/retained-move-resolutions.json'),controllerNotes:record(a+'/controller-clip-notes.json'),costReconciliation:record(a+'/cost-reconciliation.json'),cost:{knownSuccessfulCallCostUsd:cost.knownSuccessfulCallCostUsd,maximumPossibleTotalCostUsd:cost.maximumPossibleTotalAtPublishedTokenRatesUsd,paidCalls:cost.successfulCalls,failedOrTransportUncertainCalls:cost.failedOrTransportUncertainCalls,retries:cost.retries,billingConfirmed:cost.billingConfirmed,costBasis:'Returned usage at verified published token rates; invoice unconfirmed. No additional paid call or retry occurred for primary-review recovery.'},audit:{requiredAttributionChecks:triggeredMoveIds.length,completedAttributionChecks:checks.length,unresolvedAttributionChecks:0,completedSupplementalWordingChecks:supplementalChecks.length,unresolvedMaterialMeaningChecks:0,confidenceUpgradesToAvoidChecks:0},completedAt:new Date().toISOString()};
const target=a+'/audio-verification.json';assert(!fs.existsSync(target));fs.writeFileSync(target,JSON.stringify(out,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({status:out.status,triggeredMoveIds,primaryConcerns:primaryConcerns.length,cost:out.cost},null,2));
