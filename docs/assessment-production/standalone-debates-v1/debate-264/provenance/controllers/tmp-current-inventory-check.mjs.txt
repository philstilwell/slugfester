import fs from 'node:fs';
import assert from 'node:assert/strict';
import {fileRecord,serializedJson,validateStandaloneInventory,validatePrimarySpeakerScopeException} from './scripts/lib/assessment-production-standalone-debate-v1.mjs';
import {validateEditorScopedSource} from './scripts/lib/assessment-editor-scoped-source.mjs';
const a=process.argv.slice(2);assert([2,4].includes(a.length));assert.equal(a[0],'--debate');
const attempt=a.length===4?a[3]:null;if(attempt){assert.equal(a[2],'--attempt');assert(/^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(attempt));}
const read=p=>JSON.parse(fs.readFileSync(p));
const rec=read('docs/assessment-production/standalone-debates-v1/registry.json').debates.find(r=>r.debateNumber===a[1]);assert(rec);
const base=`${rec.root}/inventory${attempt?`/${attempt}`:''}`,candidate=read(`${base}/model-output.json`),auth=read(`${rec.root}/authorization.json`),source=read(`${rec.root}/source/source-lock.json`),events=read(`.assessment-cache/captions/${rec.videoId}/events.json`);
const failures=[],checks=[];
const check=(name,fn)=>{try{const result=fn();checks.push({name,status:'passed',result});}catch(e){failures.push({name,message:e.message});}};
check('source-eligible',()=>assert(Array.isArray(candidate.moves),'Worker did not supply an eligible inventory'));
check('identity',()=>{assert.equal(candidate.debateNumber,rec.debateNumber);assert.equal(candidate.debateId,rec.debateId);assert.equal(candidate.motion,auth.identity.motion);assert.deepEqual(candidate.assessedDebateWindowMs,source.participants.assessedDebateWindowMs);});
for(const m of candidate.moves??[])check(`source:${m.moveId}`,()=>{const s=m.sourceSpan;assert(Number.isInteger(s.startEvent)&&Number.isInteger(s.endEvent)&&s.startEvent>=0&&s.endEvent>=s.startEvent&&s.endEvent<events.length);s.startMs=events[s.startEvent].startMs;s.endMs=events[s.endEvent].startMs+events[s.endEvent].durationMs;s.excerpt=events.slice(s.startEvent,s.endEvent+1).map(e=>e.text).join(' ').replace(/\s+/g,' ').trim();assert(s.startMs>=source.participants.assessedDebateWindowMs.start&&s.endMs<=source.participants.assessedDebateWindowMs.end);if(s.excerpt.length>2200)assert((m.longerCompleteArgumentRationale??m.sourceSpan.longerCompleteArgumentRationale??m.sourceSpan.transportRationale??'').length>=40,'Long excerpt needs a source-specific complete-argument rationale');});
check('repository-inventory-contract',()=>validateStandaloneInventory(candidate,events));
if(auth.identity.primarySpeakerScopeException)check('primary-speaker-scope',()=>validatePrimarySpeakerScopeException({authorization:auth,sourceLock:source,inventory:candidate}));
if(auth.identity.editorApprovedScope)check('editor-approved-scope',()=>validateEditorScopedSource({authorization:auth,sourceLock:source,inventory:candidate}));
check('full-coverage',()=>{const blocks=Array.isArray(candidate.completeTranscriptCoverage)?candidate.completeTranscriptCoverage:candidate.completeTranscriptCoverage?.blocks;assert(Array.isArray(blocks)&&blocks.length);let next=0;const ids=new Set(candidate.moves.map(m=>m.moveId));for(const b of blocks){assert.equal(b.startEvent,next);assert(b.endEvent>=b.startEvent);next=b.endEvent+1;assert(Array.isArray(b.moveIds)&&b.moveIds.every(id=>ids.has(id)));assert(typeof b.reason==='string'&&b.reason.trim());}assert.equal(next,events.length);});
if(attempt)check('coverage-findings-resolved',()=>{const packet=read(`${base}/packet.json`);const findings=packet.coverageFindings??[];assert.deepEqual(candidate.coverageResolutionAudit?.map(f=>f.findingId),findings.map(f=>f.findingId));for(const f of candidate.coverageResolutionAudit){assert(Array.isArray(f.selectedMoveIds)&&f.selectedMoveIds.every(id=>candidate.moves.some(m=>m.moveId===id)));assert(typeof f.rationale==='string'&&f.rationale.length>=80);}});
const counts={total:candidate.moves?.length,pro:candidate.moves?.filter(m=>m.side==='pro').length,con:candidate.moves?.filter(m=>m.side==='con').length,sections:candidate.sections?.map(s=>({sectionId:s.sectionId,pro:candidate.moves.filter(m=>m.side==='pro'&&m.sectionId===s.sectionId).length,con:candidate.moves.filter(m=>m.side==='con'&&m.sectionId===s.sectionId).length,weightPercent:s.weightPercent}))};
const review=candidate.moves?.map(m=>({moveId:m.moveId,side:m.side,sectionId:m.sectionId,kind:m.moveKind,respondsToIds:m.respondsToIds,attribution:m.attributionConfidence,wording:m.materialWordingConfidence,characters:m.sourceSpan.excerpt?.length,events:[m.sourceSpan.startEvent,m.sourceSpan.endEvent],startSeconds:m.sourceSpan.startMs/1000,endSeconds:m.sourceSpan.endMs/1000,claim:m.claimSummary,reason:m.reasonSummary,quotes:m.quoteEligibleExactSpans}));
const audit={schemaVersion:'1.0-controller-prescore-inventory-validation',status:failures.length?'rejected-before-judgment':'mechanically-valid-semantic-review-pending',debateNumber:rec.debateNumber,debateId:rec.debateId,checkedAt:new Date().toISOString(),rawOutput:fileRecord(`${base}/model-output.json`),source:fileRecord(`.assessment-cache/captions/${rec.videoId}/events.json`),checks,failures,independentlyDerivedCounts:counts,moveReview:review,controllerChangedSemanticFields:false,controllerOnlyExpandedSourceSpanTextAndMilliseconds:true,primaryJudgmentsExecuted:false,scorePassExecuted:false};
for(const[p,v]of[[`${base}/source-expanded-candidate.json`,candidate],[`${base}/mechanical-audit.json`,audit]]){assert(!fs.existsSync(p),`Preserve ${p}`);fs.writeFileSync(p,serializedJson(v),{flag:'wx'});}
console.log(JSON.stringify({status:audit.status,failures,counts,moves:review},null,2));
